#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""mv-auto-pipeline 版本快照工具

一次进化结束时的**唯一收尾动作**。做四件事：
  1) 跑回归自检（tools/checks.txt 里的「不变式」逐条 grep，缺任何一条就拒绝发版）
  2) 往 CHANGELOG.md 追加一条进化记录（版本 / 项目 / 改了什么 / 为什么 / 验证）
  3) 打一份完整快照到 _versions/mv-auto-pipeline-V<版本>-<日期>.zip（可离线回退）
  4) git commit + git tag（精确 diff 与回退点）

用法：
  python tools/snapshot.py \\
      --version 1.3.2 --kind patch \\
      --project "《青铜信号》" \\
      --summary "一句话说明这一版干了什么" \\
      --item "改了 X：原因是 Y" --item "改了 Z：原因是 W" \\
      --verify "实测：10/10 段 PASS" \\
      --check "references/xxx.py|新函数名"
      [--add-check "references/xxx.py|新关键字"]   # 同时把新不变式写进 checks.txt
      [--dry-run] [--no-git] [--force]

  --kind patch  补丁号 +1（V1.3.1 → V1.3.2）  小修小补
  --kind minor  次版本 +1（V1.3.1 → V1.4）    铁律级 / 行为变更
  --kind major  主版本 +1（V1.3.1 → V2.0）    架构级重做

回退见 tools/rollback.py。
"""
import argparse
import datetime
import os
import re
import shutil
import subprocess
import sys
import zipfile

SKILL_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
VERSION_FILE = os.path.join(SKILL_DIR, "VERSION")
CHANGELOG = os.path.join(SKILL_DIR, "CHANGELOG.md")
CHECKS_FILE = os.path.join(SKILL_DIR, "tools", "checks.txt")
VERSIONS_DIR = os.path.join(SKILL_DIR, "_versions")

# 不进快照的内容（维护态 / 历史快照自身）
SNAP_IGNORE_DIRS = {"_versions", "__pycache__", ".git"}
SNAP_IGNORE_SUFFIX = (".pyc", ".zip")


def read_version():
    try:
        with open(VERSION_FILE, encoding="utf-8") as f:
            return f.read().strip()
    except OSError:
        return "0.0.0"


def bump(cur, kind):
    parts = (cur.split(".") + ["0", "0", "0"])[:3]
    try:
        a, b, c = (int(x) for x in parts)
    except ValueError:
        sys.exit("❌ VERSION 格式不对：%r（应为 1.3.1 这种三段式）" % cur)
    if kind == "major":
        return "%d.0" % (a + 1)
    if kind == "minor":
        return "%d.%d" % (a, b + 1)
    return "%d.%d.%d" % (a, b, c + 1)


def run_checks(checks_file, verbose=True):
    """跑不变式自检：<相对路径>|<必须出现的关键字>"""
    if not os.path.isfile(checks_file):
        if verbose:
            print("⚠️  未找到自检清单 %s，跳过自检" % checks_file)
        return []
    bad = []
    total = 0
    with open(checks_file, encoding="utf-8") as f:
        for raw in f:
            line = raw.strip()
            if not line or line.startswith("#") or "|" not in line:
                continue
            rel, kw = line.rsplit("|", 1)
            rel, kw = rel.strip(), kw.strip()
            total += 1
            fp = os.path.join(SKILL_DIR, rel)
            ok = False
            if os.path.isfile(fp):
                try:
                    with open(fp, encoding="utf-8", errors="replace") as fh:
                        ok = kw in fh.read()
                except OSError:
                    ok = False
            if not ok:
                bad.append((rel, kw))
    if verbose:
        print("🔍 回归自检：%d 条不变式，%d 条通过，%d 条丢失" %
              (total, total - len(bad), len(bad)))
        for rel, kw in bad:
            print("   ❌ %s  <- 找不到 %r" % (rel, kw))
    return bad


def add_checks(pairs):
    if not pairs:
        return 0
    os.makedirs(os.path.dirname(CHECKS_FILE), exist_ok=True)
    existing = ""
    if os.path.isfile(CHECKS_FILE):
        with open(CHECKS_FILE, encoding="utf-8") as f:
            existing = f.read()
    added = 0
    with open(CHECKS_FILE, "a", encoding="utf-8") as f:
        if existing and not existing.endswith("\n"):
            f.write("\n")
        for p in pairs:
            if p.strip() and p not in existing:
                f.write(p.strip() + "\n")
                added += 1
    return added


def prepend_changelog(version, kind, project, summary, items, verify, today):
    entry = ["## V%s · %s%s" % (version, project, ("（%s）" % today) if today else ""), ""]
    entry.append("**一句话**：%s" % summary)
    if items:
        entry.append("")
        for it in items:
            entry.append("- %s" % it)
    if verify:
        entry.append("")
        entry.append("**验证**：%s" % verify)
    entry.append("")
    entry.append("---")
    entry.append("")
    block = "\n".join(entry)

    if os.path.isfile(CHANGELOG):
        with open(CHANGELOG, encoding="utf-8") as f:
            old = f.read()
    else:
        old = ("# CHANGELOG · mv-auto-pipeline 进化史\n\n"
               "> 每一支 MV 跑完后，把本次沉淀的修复记录在这里，并打一份快照 + git tag。\n"
               "> 回退方法见 `tools/rollback.py`。**最新版本在最上面。**\n\n---\n\n")
    # 插到第一个 '## V' 之前（保持最新在顶部）；没有就追加到 header 之后
    #
    # ★ V1.0.4 幂等守卫：这一版手写了 CHANGELOG 条目，而本函数原来是**无条件**插一条
    #   → 跑一次 snapshot 就会出现两个 `## V1.0.4`。**同一个版本号写两遍就是漂移源**
    #   （drama 出过「VERSION=1.5 而正文停在 V1.3」的事故）。
    #   判据：CHANGELOG 里已经有这个版本号的段落 = 已记录过 → 不重复插。
    if re.search(r"^## V%s\b" % re.escape(version), old, re.M):
        print("ℹ CHANGELOG 里已有 V%s 的条目 → 不重复插入（幂等）" % version)
        return False
    marker = old.find("\n## V")
    if marker == -1:
        new = old.rstrip() + "\n\n" + block
    else:
        new = old[:marker + 1] + block + old[marker + 1:]
    with open(CHANGELOG, "w", encoding="utf-8") as f:
        f.write(new)
    return True


def make_snapshot(version, today):
    os.makedirs(VERSIONS_DIR, exist_ok=True)
    out = os.path.join(VERSIONS_DIR,
                       "mv-auto-pipeline-V%s-%s.zip" % (version, today.replace("-", "")))
    if os.path.exists(out):
        out = out.replace(".zip", "-%s.zip" % datetime.datetime.now().strftime("%H%M%S"))
    n = 0
    with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED, compresslevel=9) as z:
        for root, dirs, files in os.walk(SKILL_DIR):
            dirs[:] = [d for d in dirs if d not in SNAP_IGNORE_DIRS]
            for fn in files:
                if fn.endswith(SNAP_IGNORE_SUFFIX):
                    continue
                fp = os.path.join(root, fn)
                z.write(fp, os.path.relpath(fp, SKILL_DIR).replace("\\", "/"))
                n += 1
    return out, n


def git(args, check=True):
    r = subprocess.run(["git", "-C", SKILL_DIR] + args, capture_output=True, text=True)
    if check and r.returncode != 0:
        print("   ⚠️ git %s 失败：%s" % (" ".join(args), (r.stderr or r.stdout).strip()[:300]))
    return r


def main():
    ap = argparse.ArgumentParser(description="版本快照（自检 + CHANGELOG + zip + git tag）")
    ap.add_argument("--version", help="新版本号，如 1.3.2；不给则按 --kind 自增")
    ap.add_argument("--kind", default="patch", choices=["patch", "minor", "major"])
    ap.add_argument("--project", default="", help="本次测试的项目名，如 《青铜信号》")
    ap.add_argument("--summary", default="", help="一句话说明这一版干了什么（--check-only 时可省略）")
    ap.add_argument("--item", action="append", default=[], help="具体改动（可多次）")
    ap.add_argument("--verify", default="", help="验证结论")
    ap.add_argument("--add-check", action="append", default=[],
                    help="同时把新不变式写进 checks.txt（格式 相对路径|关键字）")
    ap.add_argument("--check-only", action="store_true", help="只跑自检，不发版")
    ap.add_argument("--dry-run", action="store_true", help="只打印将要做什么")
    ap.add_argument("--no-git", action="store_true", help="跳过 git 提交打标")
    ap.add_argument("--force", action="store_true", help="自检失败也强行发版")
    args = ap.parse_args()

    today = datetime.date.today().strftime("%Y-%m-%d")

    if args.add_check:
        n = add_checks(args.add_check)
        print("➕ 新增不变式 %d 条 → tools/checks.txt" % n)

    bad = run_checks(CHECKS_FILE)
    if args.check_only:
        sys.exit(1 if bad else 0)

    # 修正：--check-only 是「只跑自检」，不该被 --summary 拦住（原先 required=True
    # 导致 `snapshot.py --check-only` 直接 argparse 报错、自检根本跑不起来）。
    # 改成：只在真正发版时才要求 --summary。
    if not args.summary:
        sys.exit("❌ 发版必须提供 --summary（只跑自检请加 --check-only）")

    cur = read_version()
    new = args.version or bump(cur, args.kind)
    if new == cur:
        sys.exit("❌ 新版本号与当前一致（%s），没有变化就不该发版" % cur)

    print("📦 版本：V%s → V%s（%s）" % (cur, new, args.kind))
    if bad and not args.force:
        print("\n❌ 自检未通过：有 %d 条老修复在当前代码里找不到了。" % len(bad))
        print("   这通常意味着这次进化**把之前的修复弄丢了**。请先修好，")
        print("   确认是故意移除时再跑 --force（不推荐）。")
        sys.exit(2)

    if args.dry_run:
        print("（dry-run，不落盘）")
        return

    # 1) VERSION
    with open(VERSION_FILE, "w", encoding="utf-8") as f:
        f.write(new + "\n")
    # 2) CHANGELOG
    wrote = prepend_changelog(new, args.kind, args.project, args.summary,
                              args.item, args.verify, today)
    # 3) 快照
    zpath, zn = make_snapshot(new, today)

    print("\n✅ VERSION → %s" % new)
    print("✅ CHANGELOG.md %s V%s 条目" % ("已追加" if wrote else "已有（跳过）", new))
    print("✅ 快照 %s（%d 个文件 · %.1f KB）" %
          (os.path.basename(zpath), zn, os.path.getsize(zpath) / 1024.0))

    # 4) git
    if not args.no_git:
        if not os.path.isdir(os.path.join(SKILL_DIR, ".git")):
            git(["init", "-q"])
            print("   git 仓库已初始化")
        git(["add", "-A"])
        msg = "V%s %s: %s" % (new, args.project, args.summary)
        r = git(["commit", "-q", "-m", msg], check=False)
        if r.returncode != 0:
            print("   ⚠️ 提交失败（可能无改动）：%s" % (r.stdout + r.stderr).strip()[:200])
        tag = "V%s" % new
        t = git(["tag", tag], check=False)
        if t.returncode == 0:
            print("   ✅ git tag %s" % tag)
        else:
            print("   ⚠️ tag %s 已存在或创建失败" % tag)
        head = git(["log", "-1", "--oneline"], check=False).stdout.strip()
        print("   HEAD: %s" % head)

    print("\n🎉 已发版 V%s。回退到任意历史版本：python tools/rollback.py --list / --to V1.3.1" % new)


if __name__ == "__main__":
    main()
