"""离线单测：**凭证三级加载**（V1.0.6）—— 零算力、零网络、零真钱。

守的是什么（用户原话：「我不是说了把这些key永久记住吗，怎么还找我要？」）：
  key 记在 MEMORY.md 里没用 —— **代码不读就等于没记**。
  本单测钉死：全新空目录 / 无 .env / 不传 --api-key，
  resolve_key() 必须自己从用户级 keys.env 读到，而不是抛「缺少 key」。

跑法：python examples/test_credentials.py   （rc=0 全过）
"""
"""V1.0.6 凭证三级加载验证 —— 零算力、零网络。

核心命题（用户原话：「我不是说了把这些key永久记住吗，怎么还找我要？」）：
  在**全新空目录、无 .env、不传 --api-key** 的情况下，
  resolve_key() 必须自己从用户级 keys.env 读到 key，而不是抛「缺少 key」。
"""
import os
import shutil
import sys
import tempfile

SKILL = r"[项目工作目录]"
sys.path.insert(0, SKILL)
import rh_client as rh  # noqa: E402

NAMES = ["project_provider_API_KEY", "project_provider_API_KEY_C",
         "project_provider_API_KEY_E", "项目模型_API_KEY", "项目模型_API_BASE"]
ok = fail = 0


def check(cond, label, extra=""):
    global ok, fail
    if cond:
        ok += 1
        print(f"  PASS  {label}")
    else:
        fail += 1
        print(f"  FAIL  {label} {extra}")


def wipe_env():
    for n in NAMES:
        os.environ.pop(n, None)
    rh._INJECTED.clear()
    rh.LAST_SOURCE.clear()


print("=" * 78)
print("① 用户级凭证文件是否就位")
print("=" * 78)
print(f"CRED_ENV = {rh.CRED_ENV}")
check(os.path.isfile(rh.CRED_ENV), "用户级 keys.env 存在")
parsed = rh._read_env_file(rh.CRED_ENV, tuple(NAMES))
print(f"  解析到 {len(parsed)} 个键：{sorted(parsed)}")
check("project_provider_API_KEY" in parsed, "含消费级 project_provider_API_KEY")
check("project_provider_API_KEY_E" in parsed, "含企业级 project_provider_API_KEY_E")
check("项目模型_API_KEY" in parsed, "含 项目模型_API_KEY")
check("项目模型_API_BASE" in parsed, "含 项目模型_API_BASE")

print()
print("=" * 78)
print("② 【关键】全新空目录 / 无 .env / 不传 --api-key → 必须读得到")
print("=" * 78)
empty = tempfile.mkdtemp(prefix="mv_empty_")
old_cwd = os.getcwd()
try:
    os.chdir(empty)
    wipe_env()
    check(not os.path.isfile(os.path.join(empty, ".env")), "确认该目录确实没有 .env")
    try:
        k = rh.resolve_key()          # ← 不传任何东西
        check(bool(k), f"resolve_key() 自愈成功（key 前 8 位 {k[:8]}…）")
        check(k == parsed["project_provider_API_KEY"], "读到的正是用户级里的消费级 key")
    except SystemExit as e:
        check(False, "resolve_key() 自愈成功", f"← 竟然抛了 SystemExit：{e}")
    ke = rh.enterprise_key()
    check(ke == parsed.get("project_provider_API_KEY_E"), "enterprise_key() 也能读到")

    print()
    print("  credential_report()（不打印 key 本体）：")
    for r in rh.credential_report(empty):
        print(f"    {r['name']:<24} present={str(r['present']):<5} "
              f"source={r['source']:<12} {r['masked']}")

    print()
    print("③ 优先级：真实环境变量 > 项目 .env > 用户级 keys.env")
    # 项目级覆盖用户级
    with open(os.path.join(empty, ".env"), "w", encoding="utf-8") as f:
        f.write("# 项目级覆盖测试\nproject_provider_API_KEY=PROJECT_LEVEL_WINS\n")
    wipe_env()
    k = rh.resolve_key(project_root=empty)
    check(k == "PROJECT_LEVEL_WINS", "项目 .env 覆盖用户级", f"实际={k}")
    check(rh.LAST_SOURCE.get("project_provider_API_KEY") == "project-env",
          "来源标注 = project-env", rh.LAST_SOURCE.get("project_provider_API_KEY"))
    # 项目里没写的键仍从用户级来
    kc = os.environ.get("project_provider_API_KEY_C", "")
    check(kc == parsed["project_provider_API_KEY_C"], "项目 .env 未覆盖的键仍回落用户级")
    # 真实环境变量最高
    wipe_env()
    os.environ["project_provider_API_KEY"] = "REAL_ENV_WINS"
    k = rh.resolve_key(project_root=empty)
    check(k == "REAL_ENV_WINS", "真实环境变量压过项目 .env", f"实际={k}")
    check(rh.LAST_SOURCE.get("project_provider_API_KEY") == "real-env",
          "来源标注 = real-env", rh.LAST_SOURCE.get("project_provider_API_KEY"))
    # CLI 最高
    k = rh.resolve_key("CLI_WINS", project_root=empty)
    check(k == "CLI_WINS", "--api-key 最高优先")

    print()
    print("④ 幂等 & 不被自查污染")
    wipe_env()
    a = rh.load_dotenv(empty, quiet=True)
    b = rh.load_dotenv(empty, quiet=True)
    check(a == b, "连续两次 load 结果一致（幂等）")
    check(rh.LAST_SOURCE.get("project_provider_API_KEY") == "project-env",
          "第二次仍认得出项目级（_INJECTED 快照机制生效）",
          rh.LAST_SOURCE.get("project_provider_API_KEY"))

    print()
    print("⑤ 缺 key 时的报错文案必须指向用户级文件")
    wipe_env()
    os.remove(os.path.join(empty, ".env"))            # 项目级也拿掉
    ghost = os.path.join(empty, "nope.env")           # 指向不存在的凭证文件
    try:
        # ① 默认路径（真实存在）→ 不该抛错
        check(bool(rh.resolve_key(project_root=empty)), "默认凭证路径下能读到（未误抛）")
        # ② 把凭证路径指到不存在的地方 → 必须抛，且文案要指向该路径
        wipe_env()          # 上一步已把 key 灌进 os.environ，这里先清干净才算「真缺」
        rh.resolve_key(project_root=empty, cred_path=ghost)
        check(False, "缺 key 应抛 SystemExit")
    except SystemExit as e:
        msg = str(e)
        check(ghost in msg, "报错文案含「实际使用的」凭证路径", msg[:200].replace("\n", " "))
        check("--api-key" in msg and "用户级凭证" in msg, "报错文案给出四种给法")
        check("缺" in msg or "缺少" in msg, "报错文案明说缺 key")
    # ③ 文案里那个「标准位置」常量本身必须指向真实存在的用户级文件
    check(os.path.basename(rh.CRED_ENV) == "keys.env" and "credentials" in rh.CRED_ENV,
          "CRED_ENV 常量指向 ~/.project_provider/credentials/.../keys.env",
          rh.CRED_ENV)
finally:
    os.chdir(old_cwd)
    shutil.rmtree(empty, ignore_errors=True)

print()
print("=" * 78)
print(f"结果：{ok} PASS / {fail} FAIL")
print("=" * 78)
sys.exit(1 if fail else 0)
