def esc(x):
    return html.escape("" if x is None else str(x))


def uri(p):
    """审核页里所有图片/视频路径一律 file:/// 绝对路径（相对路径在 host 预览服务下解析不到）。"""
    try:
        ap = os.path.abspath(p).replace("\\", "/").lstrip("/")
        return "file:///" + ap
    except Exception:  # noqa: BLE001
        return p


def img_or_miss(p, label=""):
    if p and os.path.isfile(p):
        return f"<figure><img src='{esc(uri(p))}'><figcaption>{esc(label)}</figcaption></figure>"
    return f"<span class='miss'>缺图<br>{esc(label)}</span>"


def read(p, d=None):
    try:
        return json.load(open(p, encoding="utf-8"))
    except (OSError, ValueError):
        return d


def page(root, out_name, title, body, kpi: list):
    nav = (" · ".join(f"<a href='{esc(f)}'>{esc(f.split('_',1)[-1].replace('.html',''))}</a>"
                      for f in sorted(os.listdir(os.path.join(root, "审核")))
                      if f.endswith(".html"))) if os.path.isdir(os.path.join(root, "审核")) else ""
    doc = f"""<!DOCTYPE html><html lang="zh-CN"><head><meta charset="utf-8">
<title>{esc(title)}</title><style>{CSS}</style></head><body>
<header><h1>{esc(title)}</h1><div class="kpi">{''.join(kpi)}</div>
<div style="margin-top:8px;font-size:12px;color:#8a939c">{nav}</div></header>
<main>{body}</main></body></html>"""
    d = os.path.join(root, "审核")
    os.makedirs(d, exist_ok=True)
    p = os.path.join(d, out_name)
    open(p, "w", encoding="utf-8").write(doc)
    print(f"  → {p}")
    return p


def badge(state):
    m = {"approved": ("b-ok", "已通过"), "reviewed_ok": ("b-ok", "机检通过"),
         "partial": ("b-warn", "部分通过"), "pending": ("b-warn", "待审核"),
         "reviewed": ("b-warn", "已审待确认"), "draft": ("b-warn", "草稿"),
         "skipped": ("b-skip", "已跳过"), "machine_failed": ("b-bad", "机检不过"),
         "failed": ("b-bad", "失败"), "none": ("b-skip", "未审")}
    c, t = m.get(state, ("b-warn", state or "未知"))
    return f"<span class='badge {c}'>{esc(t)}</span>"

def s_lyrics(root):
    d = read(os.path.join(root, "歌词", "lyrics.json"))
    if not d:
        return page(root, "01_歌词时间轴审核.html", "关 1 · 歌词时间轴审核",
                    "<div class='box'><项目视频模型>未审</项目视频模型><p class='red'>还没有 歌词/lyrics.json</p>"
                    "<p>先跑 <code>extract_lyrics.py</code> 再跑 <code>align_lyrics.py</code>。</p></div>",
                    ["<span>状态 " + badge("none") + "</span>"])
    st = d.get("stats") or {}
    rows = []
    for r in d["lines"]:
        cls = "b-warn" if r.get("review_needed") else ""
        mark = "⚠ 待核" if r.get("review_needed") else "✅"
        mm = lambda t: f"{int(t//60)}:{t%60:05.2f}"
        rows.append(f"<tr class='{cls}'><td>{r['idx']}</td><td>{mm(r['start'])}</td>"
                    f"<td>{mm(r['end'])}</td><td>{r['end']-r['start']:.2f}s</td>"
                    f"<td>{esc(r['text'])}</td>"
                    f"<td>{r.get('hit_ratio', 1):.0%}</td><td>{mark}</td></tr>")
    body = (f"<div class='wrap'><table><thead><tr><th>#</th><th>起</th><th>止</th><th>时长</th>"
            f"<th>歌词</th><th>对齐命中</th><th>状态</th></tr></thead><tbody>{''.join(rows)}</tbody>"
            f"</table></div>"
            f"<div class='box'><项目视频模型>怎么改</项目视频模型><ul>"
            f"<li>文本不对 → 改 <code>歌词/lyrics_raw.txt</code> 重跑 align_lyrics.py</li>"
            f"<li>时间不对 → 改 <code>歌词/lyrics.lrc</code> 后 <code>align_lyrics.py --lrc ... --qc-only</code></li>"
            f"<li><b>⚠ 待核</b>行必须人工听一遍（铁律 A2：ASR 只给时间，文本必须校正）</li>"
            f"<li>确认后回对话说「歌词时间轴通过」</li></ul></div>")
    return page(root, "01_歌词时间轴审核.html", "关 1 · 歌词时间轴审核", body,
                [f"<span>曲长 <b>{esc(d.get('song_duration'))}s</b></span>",
                 f"<span>行数 <b>{len(d['lines'])}</b></span>",
                 f"<span>对齐命中率 <b>{st.get('hit_ratio', 0):.1%}</b></span>",
                 f"<span>待核行 <b class='{'red' if st.get('review_lines') else 'green'}'>"
                 f"{esc(st.get('review_lines', 0))}</b></span>"])

def s_cast(root):
    sb = read(os.path.join(root, "分镜脚本", "storyboard.json"), {}) or {}
    no_hero = (sb.get("meta") or {}).get("no_hero")
    blocks = []
    for base, kind in (("人物资产", "char"), ("场景资产", "scene"), ("道具资产", "prop")):
        d0 = os.path.join(root, base)
        if not os.path.isdir(d0):
            continue
        for d in sorted(os.listdir(d0)):
            full = os.path.join(d0, d)
            if not os.path.isdir(full):
                continue
            img = os.path.join(full, "images", "front.png") if kind == "char" \
                else os.path.join(full, "main.png")
            prof = os.path.join(full, "profile.md")
            pd = open(prof, encoding="utf-8").read()[:900] if os.path.isfile(prof) else ""
            blocks.append(f"<div class='card'><项目视频模型>{esc(d)}</项目视频模型>"
                          f"<div class='thumbs'>{img_or_miss(img, d)}</div>"
                          f"<pre>{esc(pd)}</pre></div>")
    tip = ("<div class='box'><项目视频模型>无主角模式</项目视频模型><p>本片为无主角音乐 MV，"
           "人物资产阶段跳过，画面由场景/氛围/群像承载（铁律 C6）。</p></div>") if no_hero else ""
    if no_hero:
        return page(root, "03_人物与资产审核.html", "关 3 · 人物与资产审核",
                    tip + f"<div class='grid'>{''.join(blocks) or '<div class=card>无资产</div>'}</div>",
                    ["<span>状态 " + badge("skipped") + "（无主角模式）</span>"])
    return page(root, "03_人物与资产审核.html", "关 3 · 人物与资产审核",
                f"<div class='grid'>{''.join(blocks) or '<div class=card>还没有资产</div>'}</div>",
                [f"<span>状态 {badge('approved' if blocks else 'pending')}</span>"])

def s_prompt(root):
    rv = read(os.path.join(root, "状态", "prompt_review.json"), {}) or {}
    cards = []
    for r in rv.get("segments", []):
        thumbs = "".join(img_or_miss(os.path.join(root, s["path"]),
                                     f"图片 · {esc(s.get('name'))}") for s in (r.get("slots") or []))
        ck = "".join(f"<tr><td>{esc(c['name'])}</td><td>{'✅' if c['ok'] else '⛔'}</td>"
                     f"<td>{esc(c['note'])}</td></tr>" for c in (r.get("checks") or []))
        val = ""
        for k, v in (r.get("validators") or {}).items():
            if not isinstance(v, dict):
                continue
            if v.get("skipped"):
                val += f"<div class='sub'>{k} 校验器：跳过（{esc(v.get('reason'))}）</div>"
            else:
                val += (f"<div class='sub'>{k} 校验器：错误 <b>{esc(v.get('errors'))}</b> · "
                        f"警告 {esc(v.get('warns'))}</div>")
        pp = os.path.join(root, "视频", r.get("dir", ""), "prompt.txt")
        txt = open(pp, encoding="utf-8").read() if os.path.isfile(pp) else "（缺 prompt.txt）"
        w = r.get("music_window") or {}
        cards.append(
            f"<div class='card'><项目视频模型>段 {r['seq']} {badge(r.get('status'))}</项目视频模型>"
            f"<div class='sub'>音乐窗口 {w.get('start')}–{w.get('end')}s（{w.get('dur')}s）· "
            f"口型 {esc(r.get('sing_mode'))} · 景别 {esc(r.get('shot_size'))} · "
            f"sha1 {esc(r.get('sha1'))}</div>"
            f"<div class='thumbs'>{thumbs or '<span class=miss>本段无参考图</span>'}</div>"
            f"<div class='sub'>歌词：{esc((r.get('lyric_text') or '—').replace(chr(10),' / '))}</div>"
            f"{val}<div class='wrap'><table><tbody>{ck}</tbody></table></div>"
            f"<pre>{esc(txt)}</pre></div>")
    top = (f"<div class='box'><项目视频模型>分辨率复核（只确认、不重选）</项目视频模型><ul>"
           f"<li>分镜记录档位：{esc(rv.get('resolution_desc') or '（未记录）')}</li>"
           f"<li>工作流默认分辨率值：<b>{esc((rv.get('resolution_default') or 2))}</b>（用户指定 2）</li>"
           f"<li>已确认：{'✅ ' + esc(rv.get('confirmed_resolution')) if rv.get('resolution_confirmed') else '⏳ 待确认'}</li>"
           f"</ul></div>")
    return page(root, "05_提示词审核.html", "关 5 · 视频提示词审核（硬门禁）",
                top + f"<div class='grid'>{''.join(cards) or '<div class=card>还没有提示词</div>'}</div>",
                [f"<span>段数 <b>{esc(len(rv.get('segments') or []))}</b></span>",
                 f"<span>机检通过 <b>{sum(1 for r in (rv.get('segments') or []) if r.get('status')=='reviewed_ok')}</b></span>",
                 f"<span>状态 {badge(rv.get('status') or 'none')}</span>",
                 f"<span>分辨率 {'✅ 已复核' if rv.get('resolution_confirmed') else '⏳ 未复核'}</span>"])

        w = s["music_window"]
        cards.append(f"<div class='card'><项目视频模型>段 {s['seq']} {badge(lg.get('status') or 'none')}</项目视频模型>"
                     f"<div class='sub'>窗口 {w['start']:.2f}–{w['end']:.2f}s · 目标 {w['dur']:.2f}s · "
                     f"实得 {esc(lg.get('dur') or '—')}s · {esc(lg.get('conform'))} · "
                     f"{esc(lg.get('cost_s') or '—')}s · {esc(lg.get('coins') or '—')} 币</div>"
                     f"{v}</div>")
    return page(root, "07_分段视频审核.html", "关 7 · 分段视频审核",
                f"<div class='grid'>{''.join(cards)}</div>",
                [f"<span>段数 <b>{esc(len(sb.get('segments') or []))}</b></span>"])


def s_final(root):
    fd = os.path.join(root, "视频", "final")
    rows = []
    for f in ("final.mp4", "final_master.mp4"):
        p = os.path.join(fd, f)
        if os.path.isfile(p):
            stt = os.stat(p)
            rows.append(f"<tr><td>{esc(f)}</td><td>{stt.st_size//1024//1024} MB</td>"
                        f"<td>{datetime.fromtimestamp(stt.st_mtime):%Y-%m-%d %H:%M:%S}</td>"
                        f"<td>{esc(rel_or(p, root))}</td></tr>")
    meta = ("<div class='wrap'><table><thead><tr><th>文件</th><th>大小</th><th>更新时间</th>"
            "<th>路径</th></tr></thead><tbody>" + "".join(rows) + "</tbody></table></div>") \
        if rows else "<div class='box'><项目视频模型>还没有成片</项目视频模型></div>"
    vids = "".join(f"<video controls preload='none' src='{esc(uri(os.path.join(fd, f)))}'></video>"
                   f"<div class='sub'>{esc(f)}</div>"
                   for f in ("final_master.mp4", "final.mp4") if os.path.isfile(os.path.join(fd, f)))
    va = os.path.join(fd, "对齐验收.html")
    link = f"<div class='box'><项目视频模型>对齐验收</项目视频模型><p><a href='{esc(uri(va))}'>打开 对齐验收.html</a></p></div>" \
        if os.path.isfile(va) else ""
    return page(root, "08_成片审核.html", "关 8 · 成片审核",
                vids + meta + link,
                [f"<span>成片 {badge('approved' if os.path.isfile(os.path.join(fd,'final_master.mp4')) else 'none')}</span>"])


def rel_or(p, root):
    try:
        return os.path.relpath(p, root)
    except ValueError:
        return p


def s_overview(root):
    sb = read(os.path.join(root, "分镜脚本", "storyboard.json"), {}) or {}
    ly = read(os.path.join(root, "歌词", "lyrics.json"), {}) or {}
    mp = read(os.path.join(root, "状态", "music_plan.json"), {}) or {}
    rv = read(os.path.join(root, "状态", "prompt_review.json"), {}) or {}
    st = read(os.path.join(root, "状态", "pipeline_state.json"), {}) or {}
    ledger = os.path.join(root, "状态", "成本台账.csv")
    coins = cost_s = 0
    nseg = 0
    if os.path.isfile(ledger):
        import csv as _csv
        for row in _csv.DictReader(open(ledger, encoding="utf-8-sig")):
            nseg += 1
            coins += int(row.get("供应商报告费用") or 0)
            cost_s += int(row.get("生成耗时s") or 0)
    fd = os.path.join(root, "视频", "final")
    final_p = os.path.join(fd, "final_master.mp4") if os.path.isfile(os.path.join(fd, "final_master.mp4")) \
        else os.path.join(fd, "final.mp4")
    fsz = os.path.getsize(final_p) // 1024 // 1024 if os.path.isfile(final_p) else 0
    stages = [
        ("① 歌词提取与对齐", bool(ly), f"{len(ly.get('lines') or [])} 行 → 歌词/lyrics.json"),
        ("② 风格锁定", bool(sb.get("style_prompt")), (sb.get("style_prompt") or {}).get("label", "")),
        ("③ 人物与资产", bool(os.path.isdir(os.path.join(root, "人物资产"))),
         "无主角模式（跳过）" if (sb.get("meta") or {}).get("no_hero") else "人物/场景/道具"),
        ("④ MV 分镜", bool(sb.get("segments")),
         f"{(sb.get('meta') or {}).get('segments')} 段 · 对口型 {(sb.get('meta') or {}).get('sync_ratio', 0):.0%}"),
        ("⑤ 项目视频模型 提示词", bool(rv.get("segments")), f"{len(rv.get('segments') or [])} 段 · {rv.get('status')}"),
        ("⑥ 音乐窗口核对", bool(mp), f"合计 {mp.get('window_sum')}s / 曲长 {mp.get('song_duration')}s"),
        ("⑦ 批量生成", nseg > 0, f"{nseg} 段落账 · {cost_s}s · {coins} 币"),
        ("⑧ 拼接成片", bool(fsz), f"{fsz} MB"),
        ("⑨ 对齐验收", os.path.isfile(os.path.join(fd, "对齐验收.html")), "视频/final/对齐验收.html"),
    ]
    tl = "".join(
        f"<div class='st {'ok' if ok else 'skip'}'><span class='dot'></span>"
        f"<b>{esc(n)}</b> {badge('approved' if ok else 'skipped')}"
        f"<div class='sub'>{esc(det)}</div></div>" for n, ok, det in stages)
    body = (f"<div class='box'><项目视频模型>MV 全貌</项目视频模型><ul>"
            f"<li>歌曲：{esc((sb.get('song') or {}).get('file'))} · 曲长 "
            f"<b>{esc((sb.get('song') or {}).get('duration'))}s</b></li>"
            f"<li>成片：{fsz} MB · {esc(rel_or(final_p, root))}</li>"
            f"<li>机器生成耗时合计：<b>{cost_s}s（{cost_s/3600:.2f} h）</b> —— 并发下用户等待 ≈ "
            f"{cost_s/3600/5:.2f} h</li>"
            f"<li>供应商报告费用：<b>{coins}</b>（逐段实测累加，非估算）</li></ul></div>"
            f"<div class='box'><项目视频模型>流程时间轴</项目视频模型><div class='tl'>{tl}</div></div>"
            f"<div class='box'><项目视频模型>产物文件清单</项目视频模型><ul>"
            f"<li>歌词时间轴：<code>歌词/lyrics.json</code> · <code>歌词/lyrics.lrc</code></li>"
            f"<li>分镜：<code>分镜脚本/storyboard.json</code> · <code>分镜脚本/分镜表.md</code></li>"
            f"<li>提示词：<code>视频/shot_NN_*/prompt.txt</code></li>"
            f"<li>成片：<code>视频/final/final_master.mp4</code>（交付版）· "
            f"<code>final.mp4</code>（逐段内嵌音乐版）</li></ul></div>"
            f"<div class='box'><项目视频模型>复盘三问（铁律 F1）</项目视频模型><ul>"
            f"<li>实测 vs 预设：段数 / 时长 / 币 / 耗时 差多少？</li>"
            f"<li>翻车清单：哪里报错、哪里返工、用户提了什么？</li>"

                 f"<span>生成耗时合计 <b>{cost_s}s</b></span>",
                 f"<span>供应商报告费用 <b>{coins}</b></span>"])

          "prompt": s_prompt, "music": s_music, "video": s_video, "final": s_final,
          "overview": s_overview}


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--project-root", required=True)
    ap.add_argument("--stage", default="all", choices=list(STAGES) + ["all"])
    args = ap.parse_args()
    root = os.path.abspath(args.project_root)
    todo = list(STAGES) if args.stage == "all" else [args.stage]
    print(f"生成审核页（{len(todo)} 张）→ {os.path.join(root, '审核')}")
    for s in todo:
        STAGES[s](root)
    print("完成。")


if __name__ == "__main__":
    main()
