#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""tools/scan_paths.py 的对抗测试 —— **主动伪造泄漏场景，验证它真的会拦住**

================================================================================
为什么必须有这个文件
================================================================================
只跑一次「现在扫出来是干净的」**等于没测防护** ——
它只能证明「此刻恰好没有」，证明不了「以后出现了会被拦住」。

所以这里主动构造各种「泄漏 / 不该报」的样本，逐条断言：
  · 该报的必须报（本机用户目录、POSIX 家目录、裸用户名）
  · 不该报的**绝不能**报（占位符模板、系统目录、相对路径）—— 误报会把扫描器变成噪声，
    用户学会忽略它，防护就废了
  · **扫描器不能报自己**（自指陷阱：脚本里若写着真实用户名字面量，它自己就是泄漏源）

样本路径全部**运行时拼装**，文件里不出现任何真实用户名的字面量。

用法：python examples/test_scan_paths.py
退出码：0 = 全过
"""
import os
import shutil
import sys
import tempfile
import zipfile
from pathlib import Path

SKILL = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(SKILL / "tools"))
import scan_paths as sp                                    # noqa: E402

SEP = chr(92)                                              # 反斜杠：不在源码里写出真实盘符路径
USER = (os.environ.get("USERNAME") or os.environ.get("USER") or Path.home().name).strip()
FAKE_OTHER = "someuser"                                    # 虚构用户名，不是本机真实用户

results = []


def chk(name, ok, detail=""):
    results.append((name, bool(ok), detail))


def scan_text_samples(samples):
    """把 {文件名: 内容} 落到临时目录里扫一遍。"""
    d = Path(tempfile.mkdtemp(prefix="scan_test_"))
    try:
        for fn, body in samples.items():
            (d / fn).write_text(body, encoding="utf-8")
        return sp.scan_tree(d)
    finally:
        shutil.rmtree(d, ignore_errors=True)


def hard_of(res):
    return [h for h in res["hits"] if h["level"] == "硬拦"]


def soft_of(res):
    return [h for h in res["hits"] if h["level"] == "软提示"]


# ---------------------------------------------------------------- 样本
win_user = "C:" + SEP + "Users" + SEP + USER + SEP + ".项目模型" + SEP + "skills" + SEP + "x.py"
win_other = "C:" + SEP + "Users" + SEP + FAKE_OTHER + SEP + "proj" + SEP + "a.py"
posix_home = "/home/" + USER + "/proj/a.py"
posix_mac = "/Users/" + USER + "/proj/a.py"
bare = "路径：" + USER + SEP + "abc"
ph_win = "C:" + SEP + "Users" + SEP + "<你的用户名>" + SEP + ".项目模型"
ph_win2 = "C:" + SEP + "Users" + SEP + "%USERNAME%" + SEP + "x"
ph_posix = "/home/<user>/proj"
sysdir = 'win = Path(os.environ.get("WINDIR", r"C:' + SEP + 'Windows")) / "Fonts"'
rel = "python references/deps.py\npython tools/doctor.py"
backslash_in_regex = 're.compile(r"[A-Za-z]:' + SEP + SEP + 'Users")'
# 省略号写法：文档里「你自己那台机器的用户名」的常见简写（不是泄漏）
ph_ellipsis = "C:" + SEP + "Users" + SEP + "..." + SEP + "proj"
ph_ellipsis2 = "sys.path.insert(0, r'C:" + SEP + "Users" + SEP + "..." + SEP + "skill" + SEP + "...')"
# 部署文档里那条**必须给出**的三平台模板（00-新电脑部署.md 步骤 1 原文形式）
deploy_tmpl = ("Windows :  C:" + SEP + "Users" + SEP + "<你的用户名>" + SEP + ".项目模型"
               + SEP + "skills" + SEP + "drama-auto-pipeline" + SEP)
# ★ 反向断言样本：省略号旁边**仍然**带着真实用户名 → 必须照样硬拦（证明没削弱门禁）
ell_real1 = "C:" + SEP + "Users" + SEP + USER + SEP + "..." + SEP + "x.py"
ell_real2 = "C:" + SEP + "Users" + SEP + "..." + SEP + USER + SEP + "x.py"

# ---------------------------------------------------------------- 1. 该报的必须报
res = scan_text_samples({"leak1.py": "p = r'" + win_user + "'"})
chk("本机用户目录（反斜杠）→ 硬拦", len(hard_of(res)) == 1, f"实得 {len(hard_of(res))}")

res = scan_text_samples({"leak2.py": "p = r'" + win_other + "'"})
chk("别家的 Users 目录也算本机绝对路径 → 硬拦", len(hard_of(res)) == 1, f"实得 {len(hard_of(res))}")

res = scan_text_samples({"leak3.py": "p = '" + posix_home + "'"})
chk("POSIX /home/<真实用户> → 硬拦", len(hard_of(res)) == 1, f"实得 {len(hard_of(res))}")

res = scan_text_samples({"leak4.py": "p = '" + posix_mac + "'"})
chk("macOS /Users/<真实用户> → 硬拦", len(hard_of(res)) == 1, f"实得 {len(hard_of(res))}")

res = scan_text_samples({"leak5.txt": bare})
chk("裸用户名出现在路径位置 → 硬拦", len(hard_of(res)) >= 1, f"实得 {len(hard_of(res))}")

# ★ 防削弱：加了「省略号算占位符」之后，真泄漏**绝不能**被放过
res = scan_text_samples({"leak6.py": "p = r'" + ell_real1 + "'"})
chk("省略号 + 真实用户名（用户段在前）→ 仍硬拦",
    len(hard_of(res)) >= 1, f"实得 {len(hard_of(res))}")

res = scan_text_samples({"leak7.py": "p = r'" + ell_real2 + "'"})
chk("省略号 + 真实用户名（用户段在后）→ 仍硬拦",
    len(hard_of(res)) >= 1, f"实得 {len(hard_of(res))}")

# ---------------------------------------------------------------- 2. 不该报的绝不能报
res = scan_text_samples({"tmpl.md": ph_win + "\n" + ph_win2 + "\n" + ph_posix})
chk("占位符模板（<你的用户名> / %USERNAME% / <user>）不报", len(hard_of(res)) == 0,
    f"硬拦 {len(hard_of(res))} 软提示 {len(soft_of(res))}")

res = scan_text_samples({"sys.py": sysdir})
chk("系统目录（%WINDIR% 兜底 C:\\\\Windows）不报软提示", len(soft_of(res)) == 0,
    f"实得 {len(soft_of(res))}")

res = scan_text_samples({"rel.md": rel})
chk("相对路径不报", len(res["hits"]) == 0, f"实得 {len(res['hits'])}")

res = scan_text_samples({"rx.py": backslash_in_regex})
chk("源码里的路径正则字面量不误报", len(hard_of(res)) == 0,
    f"硬拦 {len(hard_of(res))} / 软提示 {len(soft_of(res))}")

# 省略号模板：`[项目工作目录]` —— 纯点号段不带机器信息，语义上就是占位符。
# 误报这两条会把「报告里长期带着合法噪声」写死 → 用户学会忽略报告 → 防护废掉。
res = scan_text_samples({"ell.md": ph_ellipsis + "\n" + ph_ellipsis2})
chk("省略号模板（[项目工作目录] 0）",
    len(res["hits"]) == 0, f"硬拦 {len(hard_of(res))} 软提示 {len(soft_of(res))}")

res = scan_text_samples({"deploy.md": deploy_tmpl})
chk("部署文档三平台模板（[项目工作目录]",
    len(soft_of(res)) == 0, f"实得 {len(soft_of(res))}")

# ---------------------------------------------------------------- 3. 自指陷阱
res = sp.scan_tree(SKILL)
chk("扫描器不报自己（整个 skill 目录硬拦必须为 0）", len(hard_of(res)) == 0,
    "、".join(f"{h['rel']}:{h['line']}" for h in hard_of(res)[:5]) or "干净")

# ---------------------------------------------------------------- 4. zip 模式
d = Path(tempfile.mkdtemp(prefix="scan_zip_"))
try:
    leakf = d / "leak.py"
    leakf.write_text("p = r'" + win_user + "'\n", encoding="utf-8")
    cleanf = d / "clean.py"
    cleanf.write_text(rel + "\n", encoding="utf-8")
    z_dirty = d / "dirty.zip"
    with zipfile.ZipFile(z_dirty, "w") as z:
        z.write(leakf, "references/leak.py")
    z_clean = d / "clean.zip"
    with zipfile.ZipFile(z_clean, "w") as z:
        z.write(cleanf, "references/clean.py")
    r1 = sp.scan_zip(z_dirty)
    r2 = sp.scan_zip(z_clean)
    chk("zip 模式：含泄漏的包被抓到", len(hard_of(r1)) == 1, f"实得 {len(hard_of(r1))}")
    chk("zip 模式：干净的包不误报", len(hard_of(r2)) == 0, f"实得 {len(hard_of(r2))}")
finally:
    shutil.rmtree(d, ignore_errors=True)

# ---------------------------------------------------------------- 输出
print("=" * 78)
print("tools/scan_paths.py 对抗测试")
print("=" * 78)
for name, ok, detail in results:
    print(f"  [{'PASS' if ok else 'FAIL'}] {name}" + (f"  · {detail}" if detail else ""))
bad = [r for r in results if not r[1]]
print("-" * 78)
print(f"结果：{len(results) - len(bad)} 通过 / {len(bad)} 失败")
for name, _, detail in bad:
    print(f"  ✗ {name}  {detail}")
sys.exit(1 if bad else 0)
