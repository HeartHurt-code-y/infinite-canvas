#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
项目模型 图像生成 API 通用运行器（纯标准库）
提交→轮询→下载一体化。支持批量、断点续跑、失败重试。

实测数据（2026-09-12）：
- 模型 ID：zhenzhen-image-g-v2-lowprice（注意 v2，不是 v2.5）
- 单价：¥0.04/张（1k 分辨率）
- 耗时：约 30 秒/张
- 结果 URL 24h 有效

鉴权：环境变量 项目模型_project_provider_setting 或 --project_provider_setting 参数
"""

import argparse
import json
import os
import sys
import time
import urllib.request
import urllib.error
from pathlib import Path

API_BASE = "[项目内资源]"
DEFAULT_MODEL = "zhenzhen-image-g-v2-lowprice"
POLL_INTERVAL = 5  # 秒
POLL_MAX_ATTEMPTS = 60  # 最长等 5 分钟


def get_project_provider_setting(cli_key=None):
    """三级读取：--project_provider_setting > 环境变量 > 报错"""
    if cli_key:
        return cli_key
    env_key = os.environ.get("项目模型_project_provider_setting")
    if env_key:
        return env_key
    sys.exit("❌ 缺少 API Key：请设置环境变量 项目模型_project_provider_setting 或用 --project_provider_setting 传入")


# 2026-09-12 实测：Cloudflare 拦截 Python 默认 UA，必须伪装浏览器 UA，否则 403 code 1010
DEFAULT_UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
              "(KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36")


def api_call(method, path, project_provider_setting, body=None, retry=3):
    """带就地重试的 API 调用"""
    url = f"{API_BASE}{path}"
    headers = {
        "Authorization": f"Bearer {project_provider_setting}",
        "Content-Type": "application/json",
        "User-Agent": DEFAULT_UA,
    }
    data = json.dumps(body).encode("utf-8") if body else None

    last_err = None
    for attempt in range(retry):
        try:
            req = urllib.request.Request(url, data=data, headers=headers, method=method)
            with urllib.request.urlopen(req, timeout=60) as resp:
                return json.loads(resp.read().decode("utf-8"))
        except urllib.error.HTTPError as e:
            err_body = e.read().decode("utf-8", errors="replace")
            last_err = f"HTTP {e.code}: {err_body}"
            # 4xx 不重试（请求本身有问题）
            if 400 <= e.code < 500:
                raise RuntimeError(last_err)
        except (urllib.error.URLError, TimeoutError, ConnectionError) as e:
            last_err = f"{type(e).__name__}: {e}"
        # 指数退避
        wait = 2 ** attempt
        print(f"  [retry {attempt+1}/{retry}] {last_err}，{wait}s 后重试", flush=True)
        time.sleep(wait)
    raise RuntimeError(f"重试 {retry} 次仍失败：{last_err}")


def submit_image(prompt, project_provider_setting, model=DEFAULT_MODEL, resolution="1k",
                 width=None, height=None, output_format="jpeg"):
    """提交生图任务，返回 task_id"""
    body = {
        "model": model,
        "prompt": prompt,
        "metadata": {"output_format": output_format},
    }
    if resolution:
        body["metadata"]["resolution"] = resolution
    elif width and height:
        body["metadata"]["width"] = width
        body["metadata"]["height"] = height

    resp = api_call("POST", "/v1/image/generations", project_provider_setting, body)
    task_id = resp.get("task_id") or resp.get("id")
    if not task_id:
        raise RuntimeError(f"提交响应缺 task_id：{resp}")
    return task_id


def poll_task(task_id, project_provider_setting, max_attempts=POLL_MAX_ATTEMPTS):
    """轮询任务直到终态，返回完整响应"""
    for attempt in range(1, max_attempts + 1):
        resp = api_call("GET", f"/v1/image/generations/{task_id}", project_provider_setting)
        data = resp.get("data", {})
        status = data.get("status")
        progress = data.get("progress", "?")
        print(f"  [poll {attempt}] status={status} progress={progress}", flush=True)
        if status == "SUCCESS":
            return data
        if status == "FAILURE":
            reason = data.get("fail_reason", "未知")
            raise RuntimeError(f"任务失败：{reason}")
        time.sleep(POLL_INTERVAL)
    raise RuntimeError(f"轮询 {max_attempts} 次仍未完成")


def download_file(url, dest_path, retry=3):
    """下载文件，带重试"""
    dest_path = Path(dest_path)
    dest_path.parent.mkdir(parents=True, exist_ok=True)
    last_err = None
    for attempt in range(retry):
        try:
            # URL 可能含中文，转义
            safe_url = urllib.parse.quote(url, safe=":/?#[]@!$&'()*+,;=%")
            req = urllib.request.Request(safe_url, headers={"User-Agent": DEFAULT_UA})
            with urllib.request.urlopen(req, timeout=120) as resp:
                dest_path.write_bytes(resp.read())
            return dest_path
        except Exception as e:
            last_err = e
            wait = 2 ** attempt
            print(f"  [download retry {attempt+1}/{retry}] {e}，{wait}s 后重试", flush=True)
            time.sleep(wait)
    raise RuntimeError(f"下载失败：{last_err}")


def generate_one(prompt, output_path, project_provider_setting, model=DEFAULT_MODEL,
                 resolution="1k", width=None, height=None):
    """完整流程：提交→轮询→下载"""
    print(f"[submit] model={model} resolution={resolution}", flush=True)
    print(f"  prompt: {prompt[:80]}{'...' if len(prompt) > 80 else ''}", flush=True)
    task_id = submit_image(prompt, project_provider_setting, model, resolution, width, height)
    print(f"  task_id: {task_id}", flush=True)

    data = poll_task(task_id, project_provider_setting)
    result_url = data.get("result_url")
    if not result_url:
        raise RuntimeError(f"SUCCESS 但无 result_url：{data}")

    usage = data.get("usage", {})
    amount = usage.get("amount", "?")
    currency = usage.get("currency", "")
    print(f"  [done] 消耗 {amount} {currency}", flush=True)

    download_file(result_url, output_path)
    print(f"  [saved] {output_path}", flush=True)
    return {"task_id": task_id, "url": result_url, "path": str(output_path), "cost": amount}


def batch_generate(assets_file, output_dir, project_provider_setting, model=DEFAULT_MODEL,
                   resolution="1k", state_file=None):
    """
    批量生图：从资产清单 JSON 读取，逐张生成，支持断点续跑。
    资产清单格式：
    [
      {"asset_id": "char_01", "name": "苏晚", "prompt_text": "..."},
      ...
    ]
    """
    assets = json.loads(Path(assets_file).read_text(encoding="utf-8"))
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    state_path = Path(state_file) if state_file else output_dir / "_state.json"
    state = json.loads(state_path.read_text(encoding="utf-8")) if state_path.exists() else {}

    total_cost = 0.0
    for asset in assets:
        aid = asset["asset_id"]
        # 断点续跑：已 done 跳过
        if state.get(aid, {}).get("status") == "done":
            existing = state[aid].get("path")
            if existing and Path(existing).exists():
                print(f"[skip] {aid} 已完成：{existing}", flush=True)
                continue

        prompt = asset["prompt_text"]
        out_path = output_dir / f"{aid}.png"

        # 失败重试 ≤3 次
        last_err = None
        for attempt in range(3):
            try:
                print(f"\n=== [{aid}] {asset.get('name', '')} (尝试 {attempt+1}/3) ===", flush=True)
                result = generate_one(prompt, out_path, project_provider_setting, model, resolution)
                state[aid] = {"status": "done", **result}
                total_cost += float(result.get("cost") or 0)
                break
            except Exception as e:
                last_err = e
                print(f"  [fail attempt {attempt+1}] {e}", flush=True)
                time.sleep(5)
        else:
            state[aid] = {"status": "failed", "error": str(last_err)}
            print(f"  [FAILED] {aid} 重试 3 次仍失败", flush=True)

        # 每完成一张就写状态（断点保护）
        state_path.write_text(json.dumps(state, ensure_ascii=False, indent=2), encoding="utf-8")

    # 汇总
    done = sum(1 for v in state.values() if v.get("status") == "done")
    failed = sum(1 for v in state.values() if v.get("status") == "failed")
    print(f"\n=== 批量完成 ===", flush=True)
    print(f"  成功：{done} / 失败：{failed} / 总成本：¥{total_cost:.4f}", flush=True)
    print(f"  状态文件：{state_path}", flush=True)
    return state


def main():
    ap = argparse.ArgumentParser(description="项目模型 图像生成 API 运行器")
    ap.add_argument("--prompt", help="单张生图的提示词")
    ap.add_argument("--prompt-file", help="从文件读提示词（中文推荐）")
    ap.add_argument("--output", help="单张生图的输出路径")
    ap.add_argument("--batch", help="批量模式：资产清单 JSON 路径")
    ap.add_argument("--output-dir", help="批量模式：输出目录")
    ap.add_argument("--state-file", help="批量模式：状态文件路径（默认 <output_dir>/_state.json）")
    ap.add_argument("--project_provider_setting", help="API Key（默认读环境变量 项目模型_project_provider_setting）")
    ap.add_argument("--model", default=DEFAULT_MODEL, help=f"模型 ID（默认 {DEFAULT_MODEL}）")
    ap.add_argument("--resolution", default="1k", help="分辨率：1k / 2k（默认 1k）")
    ap.add_argument("--width", type=int, help="自定义宽度（与 resolution 互斥）")
    ap.add_argument("--height", type=int, help="自定义高度")
    args = ap.parse_args()

    project_provider_setting = get_project_provider_setting(args.project_provider_setting)

    if args.batch:
        if not args.output_dir:
            sys.exit("❌ 批量模式必须指定 --output-dir")
        batch_generate(args.batch, args.output_dir, project_provider_setting, args.model,
                       args.resolution, args.state_file)
    else:
        prompt = args.prompt
        if args.prompt_file:
            prompt = Path(args.prompt_file).read_text(encoding="utf-8").strip()
        if not prompt:
            sys.exit("❌ 必须指定 --prompt 或 --prompt-file")
        if not args.output:
            sys.exit("❌ 必须指定 --output")
        generate_one(prompt, args.output, project_provider_setting, args.model,
                     args.resolution, args.width, args.height)


if __name__ == "__main__":
    main()
