def page_shell(title, project_name, stage_label, body_html, audit_state="pending"):
    """通用页面骨架"""
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    badge_cls = "approved" if audit_state == "approved" else "pending"
    badge_txt = "✅ 已通过" if audit_state == "approved" else "⏳ 待审核"
    return f"""<!DOCTYPE html>
<html lang="zh-CN">
<head>
<meta charset="UTF-8">
<title>{html.escape(title)} · {html.escape(project_name)}</title>
<style>{BASE_CSS}</style>
</head>
<body>
<div class="head">
  <div>
    <h1>{html.escape(title)}</h1>
    <div class="meta">项目：{html.escape(project_name)} · 阶段：{html.escape(stage_label)} · 生成于 {now}</div>
  </div>
  <span class="badge {badge_cls}">{badge_txt}</span>
</div>
{body_html}
</body>
</html>"""


def actions_block(hints):
    """底部操作说明"""
    items = "".join(f"<p>• {h}</p>" for h in hints)
    return f"""<div class="actions">
<h4>📋 如何操作</h4>
{items}
<p style="margin-top:14px;color:var(--dim);font-size:12px">本页为静态展示，所有操作请回到 项目模型 对话里告知主 Agent。</p>
</div>"""


def _rel(p, project_root):
    """把项目内的绝对路径转成 file:/// URI。
    审核页经常被 host 预览服务以 [项目内资源] 形式加载，
    相对路径 `../场景资产/...` 在那种协议下解析不到本地文件 → 一律用 file:/// 绝对路径兜底。
    """
    try:
        return Path(p).resolve().as_uri()
    except Exception:
        return str(p)


def _stage_state(project_root, stage):
    """读某一道关的真实审核状态（V1.3：审核页徽章不再永远显示「待审核」）。

    - prompt 关读 `状态/prompt_review.json` 的 status（approved / partial / pending）
    - style 关没有独立 stage 键，以 `style_prompt.locked` 为准
    - 其余读 `状态/pipeline_state.json` 的 `stages.<stage>.status`
    """
    project_root = Path(project_root)
    if stage == "prompt":
        pr = project_root / "状态" / "prompt_review.json"
        if pr.exists():
            try:
                return json.loads(pr.read_text(encoding="utf-8")).get("status", "pending")
            except Exception:
                return "pending"
        return "pending"
    st_file = project_root / "状态" / "pipeline_state.json"
    if not st_file.exists():
        return "pending"
    try:
        st = json.loads(st_file.read_text(encoding="utf-8"))
    except Exception:
        return "pending"
    if stage == "style":
        return "approved" if (st.get("style_prompt") or {}).get("locked") else "pending"
    return ((st.get("stages", {}) or {}).get(stage, {}) or {}).get("status", "pending")


# ========== 01 剧本审核 ==========

def build_script_review(project_root):
    project_root = Path(project_root)
    units_file = project_root / "剧本" / "units_30s.json"
    if not units_file.exists():
        return None, "units_30s.json 不存在"

    data = json.loads(units_file.read_text(encoding="utf-8"))
    units = data.get("units", []) or data.get("segments", [])
    state = data.get("status", "draft")

    # 读原文
    novel_file = project_root / "剧本" / "小说原文.md"
    novel_text = ""
    if novel_file.exists():
        novel_text = novel_file.read_text(encoding="utf-8")

    # 从原文提取一句话总结（取首段或前 80 字）
    one_line = ""
    for line in novel_text.splitlines():
        line = line.strip().lstrip("#").strip()
        if line and not line.startswith(">") and not line.startswith("---"):
            one_line = line
            break
    if len(one_line) > 80:
        one_line = one_line[:80] + "..."

    # 单元逻辑分组（只读展示）
    units_summary = ""
    for u in units:
        units_summary += f"""
<div class="card">
  <h3>{html.escape(u.get('unit_id', ''))} · {html.escape(u.get('title', ''))}</h3>
  <div class="field">{html.escape(u.get('summary', ''))}</div>
  <div class="field" style="color:var(--dim);font-size:12px">场景 {html.escape(u.get('scene','-'))} · 出场 {html.escape(','.join(u.get('characters',[])))} · 约 {u.get('estimated_lines','?')} 字 / {u.get('estimated_duration_sec','?')}s</div>
</div>"""

    body = f"""
<div class="sect">
  <h2>📖 一句话总结</h2>
  <div class="card" style="border-left:3px solid var(--gold);padding:18px 22px">
    <p style="font-size:18px;line-height:1.8;color:var(--txt)">{html.escape(one_line)}</p>
  </div>
</div>
<div class="sect">
  <h2>📖 详细剧本</h2>
  <div class="card"><pre style="white-space:pre-wrap;line-height:1.9;font-size:14px">{html.escape(novel_text)}</pre></div>
</div>
<div class="sect">
  <h2>🎬 剧情单元逻辑分组（仅供参考，非硬约束）</h2>
  <p style="color:var(--mut);font-size:13px;margin-bottom:14px">单元 = 一个关系变化；H3 段切分按镜头/台词/情绪节拍自由切，不受单元时长限制</p>
  <div class="grid" style="grid-template-columns:repeat(2,1fr)">{units_summary}</div>
</div>
{actions_block([
  "若整体通过，对话里说 <code>剧本通过</code>",
  "若要改某单元，说 <code>改单元 N：xxx</code>",
  "若要重写某单元，说 <code>重写单元 N：xxx</code>",
])}
"""
    out = project_root / "审核" / "01_剧本审核.html"
    out.parent.mkdir(parents=True, exist_ok=True)
    _as = _stage_state(project_root, "script")
    if _as == "pending":
        _as = state
    out.write_text(page_shell("01 · 剧本审核", project_root.name,
                              "剧本共创", body, _as), encoding="utf-8")
    return out, None


# ========== 02 风格审核（关 2 · V1.3 新增） ==========

def _load_style_options(project_root):
    """读风格候选。

    优先 `状态/style_options.json`（V1.3 起要求落盘 3 个候选方向）：
        {"options":[{option,label,ref_films,shot_language,palette,fit_scenes,prompt_en}, ...],
         "locked_choice":"A"}
    没有该文件时退化为只用 pipeline_state.json.style_prompt 渲染「已锁定方向」。
    """
    opt_file = project_root / "状态" / "style_options.json"
    locked = None
    state_file = project_root / "状态" / "pipeline_state.json"
    if state_file.exists():
        try:
            st = json.loads(state_file.read_text(encoding="utf-8"))
            locked = st.get("style_prompt") or None
        except Exception:
            locked = None

    if opt_file.exists():
        data = json.loads(opt_file.read_text(encoding="utf-8"))
        opts = data.get("options", []) or []
        choice = data.get("locked_choice") or (locked or {}).get("choice")
        if opts:
            return opts, choice, locked, True
    # 退化：只有锁定方向
    if locked:
        return [{
            "option": locked.get("choice", "A"),
            "label": locked.get("label", "已锁定风格"),
            "ref_films": locked.get("ref_films", "—"),
            "shot_language": locked.get("shot_language", "—"),
            "palette": locked.get("palette", "—"),
            "fit_scenes": locked.get("fit_scenes", "—"),
            "prompt_en": locked.get("prompt_en", ""),
        }], locked.get("choice", "A"), locked, False
    return [], None, None, False


def build_style_review(project_root):
    project_root = Path(project_root)
    opts, choice, locked, full = _load_style_options(project_root)
    if not opts:
        return None, "风格未锁定（pipeline_state.json 缺 style_prompt，且无 style_options.json）"

    cards = ""
    for o in opts:
        key = str(o.get("option", "")).strip()
        is_sel = (key == str(choice or "").strip())
        border = "brand" if is_sel else "line"
        sel_badge = ('<span class="badge approved" style="margin-left:8px">✅ 你已选定</span>'
                     if is_sel else '<span class="badge pending" style="margin-left:8px">备选</span>')
        prompt_en = o.get("prompt_en") or ""
        pre = (f'<pre>{html.escape(prompt_en)}</pre>' if prompt_en
               else '<div class="field" style="color:var(--dim)">（未落盘 prompt_en）</div>')
        cards += f"""
<div class="card" style="border-color:{('var(--gold)' if is_sel else 'var(--line)')};{'box-shadow:0 0 0 1px rgba(245,182,66,.35)' if is_sel else ''}">
  <h3>方向 {html.escape(key)} · {html.escape(str(o.get('label','')))} {sel_badge}</h3>
  <div class="field"><b>参考片</b>：{html.escape(str(o.get('ref_films','—')))}</div>
  <div class="field"><b>镜头语言</b>：{html.escape(str(o.get('shot_language','—')))}</div>
  <div class="field"><b>调色板</b>：{html.escape(str(o.get('palette','—')))}</div>
  <div class="field"><b>适合场景</b>：{html.escape(str(o.get('fit_scenes','—')))}</div>
  <div class="field" style="margin-top:8px"><b>风格提示词（英文，全片统一）</b></div>
  {pre}
</div>"""

    note = ""
    if not full:
        note = """<div class="card" style="border-color:rgba(245,182,66,.45);background:rgba(245,182,66,.06)">
  <div class="field" style="color:var(--gold)">⚠️ 本项目的备选方向 B/C 未落盘（V1.3 之前没有 `状态/style_options.json` 这一步），本页仅记录<b>已锁定方向</b>。V1.3 起已要求把 3 个候选方向写入 `状态/style_options.json`，后续项目三张卡片都会齐。</div>
</div>"""

    locked_at = (locked or {}).get("locked_at", "—")
    body = f"""
<div class="sect">
  <h2>🎨 风格锁定（关 2 · 先于资产生成）</h2>
  <p style="color:var(--mut);font-size:13px;margin-bottom:14px">
    角色图的服装色调、场景图的光线氛围、道具图的材质表现<b>全部依赖风格锚定</b>；
    风格一旦锁定，全片统一，<b>不可段间修改</b>（铁律 B1）。
  </p>
  <div class="grid" style="grid-template-columns:repeat(auto-fill,minmax(360px,1fr))">{cards}</div>
  {note}
  <div class="card" style="margin-top:16px">
    <div class="field"><b>当前锁定</b>：方向 {html.escape(str(choice or '—'))} · {html.escape(str((locked or {}).get('label','—')))}</div>
    <div class="field"><b>锁定时点</b>：{html.escape(str(locked_at))}</div>
  </div>
</div>
{actions_block([
  "若要确认锁定，对话里说 <code>风格确认，用 A</code>",
  "若要换一组推荐，说 <code>换一组风格推荐</code>",
  "若已锁定但想改，说 <code>改风格为 xxx</code>（会作废已生成的资产）",
])}
"""
    out = project_root / "审核" / "02_风格审核.html"
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(page_shell("02 · 风格审核（关 2 · 先于资产）", project_root.name,
                              "风格锁定", body,
                              _stage_state(project_root, "style")), encoding="utf-8")
    return out, None

def build_asset_review(project_root):
    project_root = Path(project_root)

    # 角色卡片
    chars_html = ""
    chars_dir = project_root / "角色资产"
    if chars_dir.exists():
        for char_dir in sorted(chars_dir.iterdir()):
            if not char_dir.is_dir():
                continue
            char_id = char_dir.name.split("_")[0] + "_" + char_dir.name.split("_")[1] if "_" in char_dir.name else char_dir.name
            front = char_dir / "images" / "front.png"
            voice = char_dir / "voice" / "voice_sample.mp3"
            profile = char_dir / "profile.md"
            voice_locked = char_dir / "voice" / "voice_locked.json"

            profile_text = profile.read_text(encoding="utf-8")[:500] if profile.exists() else ""
            voice_info = ""
            if voice_locked.exists():
                vl = json.loads(voice_locked.read_text(encoding="utf-8"))
                voice_info = f"<div class='field'><b>音色：</b>{vl.get('method','?')} / {'🔒 已锁定' if vl.get('locked') else '未锁定'}</div>"

            img_html = f'<img src="{_rel(front, project_root)}" />' if front.exists() else "<div class='field'>（无图）</div>"
            audio_html = f'<audio controls src="{_rel(voice, project_root)}"></audio>' if voice.exists() else ""

            chars_html += f"""
<div class="card">
  <h3>{html.escape(char_dir.name)}</h3>
  {img_html}
  {audio_html}
  {voice_info}
  <pre>{html.escape(profile_text)}</pre>
</div>"""

    # 场景卡片
    scenes_html = ""
    scenes_dir = project_root / "场景资产"
    if scenes_dir.exists():

            desc = scene_dir / "description.md"

            scenes_html += f"""
<div class="card">
  <h3>{html.escape(scene_dir.name)}</h3>
  {img_html}
  <pre>{html.escape(desc_text)}</pre>
</div>"""

    # 道具卡片（SKILL.md 4.4 要求：角色卡片墙 + 场景卡片墙 + 道具卡片墙，缺一不可）
    props_html = ""
    props_dir = project_root / "道具资产"
    if props_dir.exists():
        for prop_dir in sorted(props_dir.iterdir()):
            if not prop_dir.is_dir():
                continue
            main_img = prop_dir / "main.png"
            desc = prop_dir / "description.md"
            desc_text = desc.read_text(encoding="utf-8")[:300] if desc.exists() else ""
            img_html = f'<img src="{_rel(main_img, project_root)}" />' if main_img.exists() else "<div class='field'>（无图）</div>"
            props_html += f"""
<div class="card">
  <h3>{html.escape(prop_dir.name)}</h3>
  {img_html}
  <pre>{html.escape(desc_text)}</pre>
</div>"""

    body = f"""
<div class="sect">
  <h2>🧑 角色资产（三件套：图 + 音色 + 档案）</h2>
  <div class="grid">{chars_html or '<div class="card">暂无角色</div>'}</div>

  <div class="grid">{scenes_html or '<div class="card">暂无场景</div>'}</div>

  "若全部通过，对话里说 <code>资产通过</code>",
  "若要重生某角色音色，说 <code>重生 char_01 音色：更甜一点</code>",
  "若要替换某角色图，说 <code>替换 char_01 图：我上传一张</code>",
  "若要改角色档案描述，说 <code>改 char_01 档案：xxx</code>",

def _load_style_prompt_dict(project_root):
    """读 状态/pipeline_state.json 的 style_prompt 字典（V2.0.3）。
    审核页的风格中文标签必须来自**本项目**的锁定记录，不能硬编码。"""
    f = Path(project_root) / "状态" / "pipeline_state.json"
    if not f.exists():
        return {}
    try:
        data = json.loads(f.read_text(encoding="utf-8"))
    except Exception:
        return {}
    sp = data.get("style_prompt")
    return sp if isinstance(sp, dict) else {}


def _seg_mood(s, mood_map):
    """段情绪节拍（V2.0.3）：优先读段内 mood_zh；否则查 kw 映射；再否则 '—'。
    旧版硬编码了一部别的剧的 kw→情绪表，换项目后 17 段全显示 '—'。"""
    m = s.get("mood_zh") or s.get("mood")
    if m:
        return m
    kw = s.get("kw", "")
    if kw in mood_map:
        return mood_map[kw]
    return "—"


def build_storyboard_review(project_root):
    project_root = Path(project_root)
    sb_dir = project_root / "分镜脚本"
    if not sb_dir.exists():
        return None, "分镜脚本目录不存在"

    # 读 storyboard.json（V0.4 主结构）
    sb_json_file = sb_dir / "storyboard.json"
    storyboard = {}
    segs = []
    style_prompt = ""
    if sb_json_file.exists():
        storyboard = json.loads(sb_json_file.read_text(encoding="utf-8"))
        segs = storyboard.get("segments", [])
        style_prompt = storyboard.get("style_prompt", "")

    # 读角色/场景/道具元数据（用于显示友好名）
    char_names = {}
    for d in (project_root / "角色资产").glob("char_*"):
        if d.is_dir():
            cid = d.name.split("_")[0] + "_" + d.name.split("_")[1]
            cname = d.name.split("_", 2)[-1] if "_" in d.name else d.name
            char_names[cid] = cname
    scene_names = {}
    for d in (project_root / "场景资产").glob("scene_*"):
        if d.is_dir():
            sid = d.name.split("_")[0] + "_" + d.name.split("_")[1]
            sname = d.name.split("_", 2)[-1] if "_" in d.name else d.name
            scene_names[sid] = sname
    prop_names = {}
    for d in (project_root / "道具资产").glob("prop_*"):
        if d.is_dir():
            pid = d.name.split("_")[0] + "_" + d.name.split("_")[1]
            pname = d.name.split("_", 2)[-1] if "_" in d.name else d.name
            prop_names[pid] = pname

    # 顶部：风格提示词卡（全片统一，锁定）—— 中文说明 + 英文 prompt
    # V2.0.3 修复：原先硬编码了上一个项目的风格文字（是枝裕和/富士 400H），
    # 换项目后会给用户看**错误的风格说明**。改为从 状态/pipeline_state.json
    # 的 style_prompt 字典读中文标签；缺字段时才回退到 storyboard 里的英文字符串。
    style_html = ""
    sp = _load_style_prompt_dict(project_root)
    if style_prompt or sp:
        en_prompt = sp.get("prompt_en") or (style_prompt if isinstance(style_prompt, str) else "")
        rows = []
        if sp.get("label"):
            rows.append(('<b>风格定位：</b>', sp["label"]))
        if sp.get("ref_films"):
            rows.append(('<b>参考影调：</b>', sp["ref_films"]))
        if sp.get("shot_language"):
            rows.append(('<b>镜头语言：</b>', sp["shot_language"]))
        if sp.get("palette"):
            rows.append(('<b>色彩与光线：</b>', sp["palette"]))
        if not rows and en_prompt:
            rows.append(('<b>风格定位：</b>', "（本项目未落盘中文风格标签，仅存英文 prompt，见下方）"))
        rows_html = "\n  ".join(
            '<div class="field">%s%s</div>' % (k, html.escape(str(v))) for k, v in rows)
        details_html = ""
        if en_prompt:
            details_html = f"""
  <details style="margin-top:10px">
    <summary style="cursor:pointer;color:var(--cyan);font-size:12px">查看 H3 提示词原文（英文，仅展示）</summary>
    <pre style="white-space:pre-wrap;font-size:12px;line-height:1.8;margin-top:8px">{html.escape(en_prompt)}</pre>
  </details>"""
        lock_note = ""
        if sp.get("locked") or style_prompt:
            lock_note = (
                '<p style="color:var(--dim);font-size:12px;margin-top:8px">'
                '如需修改，对话里说 <code>改风格：xxx</code>，主 Agent 会更新所有段</p>')
        style_html = f"""
<div class="card" style="border-left:3px solid var(--gold);margin-bottom:24px">
  <h3>🎨 全片风格（🔒 已锁定，全片统一 · 铁律 B1）</h3>
  {rows_html}{details_html}
  {lock_note}
</div>"""

    # 每段导演分镜卡（中文叙述）
    seg_cards = ""
    for s in segs:
        seq = s.get("seq", "?")
        kw = s.get("kw", "")
        dur = s.get("dur", "?")
        speaker = s.get("speaker_char_id", "")
        speaker_name = char_names.get(speaker, speaker)
        on_screen = [char_names.get(c, c) for c in s.get("on_screen_char_ids", [])]
        scene_id = s.get("scene_id", "")
        scene_name = scene_names.get(scene_id, scene_id)
        prop_ids = [prop_names.get(p, p) for p in s.get("prop_ids", [])]
        crowd = bool(s.get("crowd"))
        crowd_note = s.get("crowd_note") or ""
        crowd_txt = ('有（背景虚化、不开口、与女主同款校服）' if crowd
                     else ('无' + (' —— ' + html.escape(crowd_note) if crowd_note else '')))
        costume_notes = s.get("costume_notes") or {}
        costume_txt = "；".join(
            "%s %s" % (char_names.get(k, k), v) for k, v in costume_notes.items()) if isinstance(
                costume_notes, dict) else ""
        narration = s.get("narration", "")
        # V0.5.5：多角色对白逐句展示（谁说话谁开口）；无 dialogue 时退回单说话人
        dlg = s.get("dialogue")
        if isinstance(dlg, list) and dlg:
            rows = []
            tot = 0
            for d in dlg:
                cid = d.get("char_id") if isinstance(d, dict) else ""
                txt = (d.get("text") if isinstance(d, dict) else str(d)) or ""
                nm = char_names.get(cid, cid or "?")
                tot += len(txt)
                rows.append(
                    f'<p style="font-size:15px;line-height:1.8;color:var(--txt)">'
                    f'<b>{html.escape(str(nm))}</b>："{html.escape(txt)}"</p>')
            dlg_html = "\n    ".join(rows)
            dlg_count = len(dlg)

        else:
            dlg_html = (f'<p style="font-size:15px;line-height:1.8;color:var(--txt)">'

            dlg_count = 1

            dlg_title = ""
        action = s.get("action_zh") or s.get("action", "")  # 优先中文
        camera_lock = s.get("camera_lock_zh") or s.get("camera_lock", "")

        # 生成出来还是旧台词）。所以这里收的是 dialogue，导出时同步 beats + narration。
        dlg_rows_edit = []
        for d in (dlg if isinstance(dlg, list) else []):
            cid = d.get("char_id") if isinstance(d, dict) else ""
            txt = (d.get("text") if isinstance(d, dict) else "") or ""

        link = s.get("link", "origin")
        soundscape = s.get("soundscape_zh") or s.get("soundscape", "")

        # 【节拍】有序展开（V2.0.3 补齐铁律 5.3 要求的字段）
        beats = s.get("beats")

        if isinstance(beats, list) and beats:
            for i, b in enumerate(beats, 1):

                    continue
                btype = b.get("type", "action")
                if btype == "action":
                    txt = b.get("zh") or b.get("en") or ""
                    tag, col = "动作", "var(--cyan)"
                else:
                    cid = b.get("char_id", "")
                    nm = char_names.get(cid, cid or "?")
                    txt = f'{nm}："{b.get("text", "")}"'
                    if btype == "line_offscreen":
                        tag, col = "画外音", "var(--vio)"
                    elif btype == "inner":
                        tag, col = "内心独白", "var(--vio)"
                    elif btype == "narration":

        mood_map = {

            "伞歪了": "默契 / 靠近",
            "你都湿了": "发现 / 心疼",
            "屋檐下": "靠近 / 暗涌",
            "热的": "害羞 / 慌乱",
            "背后教写字": "亲密 / 心动",
            "手抖": "慌乱 / 崩溃",
            "食堂": "暧昧 / 试探",

        mood = _seg_mood(s, mood_map)

        acting = s.get("acting", {})
        acting_goal = acting.get("goal", "—")
        acting_obstacle = acting.get("obstacle", "—")
        acting_tactic = acting.get("tactic", "—")
        acting_beat = acting.get("beat_change", "—")

        acting_json = json.dumps(acting, ensure_ascii=False, indent=2) if isinstance(
            acting, dict) else "{}"

        seg_cards += f"""

  <h3>段 {seq} · {html.escape(kw)} <span class="badge side">{dur}s</span> <span class="badge" style="background:rgba(77,216,230,.15);color:{link_color};border-color:{link_color};font-size:11px;padding:3px 10px">{link_label}</span></h3>

        # 参考图槽位（顺序 = <Picture N> 编号顺序，与实跑上传顺序同源）

                       f'<div style="font-size:10px;color:var(--dim);margin-top:3px">'
                       f'{html.escape(str(rf.get("kind","")))} / {html.escape(str(rf.get("id","")))}</div></div>')

        for i, au in enumerate(r.get("audios", []), start=1):

            ap_html = (f'<audio controls src="{_rel(ap, project_root)}" style="width:100%"></audio>'

            auds += (f'<div style="flex:1 1 220px">'
                     f'<div style="font-size:11px;color:var(--gold);font-weight:700;margin-bottom:4px">'
                     f'@音色{i} · {html.escape(str(au.get("name","")))} '
                     f'<span style="color:var(--dim)">({html.escape(str(au.get("char_id","")))})</span></div>'
                     f'{ap_html}</div>')
        audio_html = (f'<div style="display:flex;gap:12px;flex-wrap:wrap;margin:10px 0">{auds}</div>'
                      if auds else
                      '<div class="field" style="color:var(--dim)">本段无音色槽（无角色开口）</div>')

        # 机检明细（铁律 D5 五条 + 结构自检）
        ma = r.get("machine_audit") or {}
        rows = ""
        for it in ma.get("items", []):
            ok = it.get("pass")
            rows += (f'<tr><td>{"✅" if ok else "❌"}</td>'
                     f'<td style="color:var(--dim);font-size:12px">{html.escape(str(it.get("code","")))}</td>'

                     f'<td style="color:var(--mut);font-size:12px">{html.escape(str(it.get("detail","")))}</td></tr>')
        val = ma.get("validator") or {}
        if val.get("errors") is not None:
            vcls = "ok" if val["errors"] == 0 else "fail"
            vtxt = (f'错误 {val.get("errors")} · 警告 {val.get("warnings")} · '
                    f'提示 {val.get("infos")}')
        else:
            vcls = "fail"
            vtxt = "校验器未运行：" + str(val.get("raw", ""))[:160]
        machine_pass = ma.get("pass")
        mp_badge = ('<span class="badge approved" style="font-size:11px;padding:3px 10px">机检✅</span>'
                    if machine_pass else
                    '<span class="badge pending" style="font-size:11px;padding:3px 10px;'
                    'background:rgba(255,92,138,.15);color:var(--mag);border-color:var(--mag)">机检❌</span>')

        # 对白行
        dlg_rows = ""
        for d in r.get("dialogue", []):
            dlg_rows += (f'<p style="font-size:14px;line-height:1.8;color:var(--txt);margin:3px 0">'
                         f'<b style="color:var(--gold)">{html.escape(str(d.get("char_id","")))}</b>：'
                         f'"{html.escape(str(d.get("text","")))}"</p>')

        # 语义审（主 Agent 用 Task 工具判剧本贴合度后回填）
        sa = r.get("subagent_audit")
        if isinstance(sa, dict):
            sa_badge = ('<span class="badge approved" style="font-size:11px;padding:3px 10px">语义审✅</span>'
                        if sa.get("pass") else
                        '<span class="badge pending" style="font-size:11px;padding:3px 10px;'
                        'background:rgba(255,92,138,.15);color:var(--mag);border-color:var(--mag)">语义审❌</span>')
            sa_txt = html.escape(str(sa.get("detail") or sa.get("note") or ""))
        else:
            sa_badge = ('<span class="badge" style="font-size:11px;padding:3px 10px;'
                        'background:rgba(95,107,130,.15);color:var(--dim);border-color:var(--dim)">语义审⏳</span>')
            sa_txt = "语义审结论待主 Agent 用 Task 工具判定后回填（判剧本贴合度 / 节奏合理性）"

        prompt_file = Path(str(r.get("shot_dir") or "")) / "prompt.txt"
        prompt_text = prompt_file.read_text(encoding="utf-8") if prompt_file.exists() else ""
        safe_prompt = html.escape(prompt_text)

        cards += f"""
<div class="card" data-seq="{seq}">
  <h3>段 {seq} · {html.escape(str(r.get("kw","")))} <span class="badge side">{r.get("dur")}s</span>
    <span class="badge" style="background:rgba(77,216,230,.15);color:{link_color};border-color:{link_color};font-size:11px;padding:3px 10px">{link_label}</span>
    <span class="badge" style="background:rgba(255,255,255,.04);color:{take_color};border-color:{take_color};font-size:11px;padding:3px 10px">{take_label}</span>
    {st_badge} {mp_badge} {sa_badge}</h3>

  <div class="field"><b>🏞 场景：</b>{html.escape(str(r.get("scene_label") or r.get("scene_id") or "—"))}
    · <b>说话人：</b>{html.escape("、".join(r.get("speakers") or []) or "无")}
    · <b>时长：</b>{r.get("dur")}s</div>

  <div style="font-size:12px;color:var(--cyan);font-weight:700;margin:12px 0 6px">参考图槽位（= &lt;Picture N&gt; 编号顺序）</div>
  <div style="display:flex;gap:10px;flex-wrap:wrap">{thumbs or '<span style="font-size:12px;color:var(--dim)">无参考图</span>'}</div>

  <div style="font-size:12px;color:var(--gold);font-weight:700;margin:14px 0 6px">音色槽位（= &lt;Audio N&gt; 编号顺序）</div>
  {audio_html}

  <div style="font-size:12px;color:var(--vio);font-weight:700;margin:14px 0 6px">台词（{r.get("dialogue_chars","?")} 字 ≈ 需 {r.get("budget_secs","?")}s / 本段 {r.get("dur")}s）</div>
  {dlg_rows or '<p style="font-size:13px;color:var(--dim)">本段无台词</p>'}

  <div style="font-size:12px;color:var(--mut);font-weight:700;margin:16px 0 6px">机检明细（铁律 D5 五条 + 结构自检）</div>
  <table><thead><tr><th></th><th>编号</th><th>检查项</th><th>结论</th></tr></thead><tbody>{rows}</tbody></table>
  <div class="field" style="margin-top:8px"><b>官方校验器（h3-ref2va）：</b><span class="{vcls}">{html.escape(vtxt)}</span></div>
  <div class="field" style="color:var(--dim);font-size:12px"><b>语义审：</b>{sa_txt}</div>

  <details style="margin-top:14px">
    <summary style="cursor:pointer;color:var(--cyan);font-size:13px;font-weight:700">📄 查看/编辑 H3 视频提示词（{len(prompt_text)} 字符 · sha1 {html.escape(str(r.get("prompt_sha1","")))}）</summary>
    <textarea id="prompt_{seq}" style="width:100%;min-height:320px;background:rgba(0,0,0,.35);border:1px solid var(--line);border-radius:8px;padding:12px;font-size:11px;line-height:1.6;color:#a8b4c9;font-family:Consolas,monospace;margin-top:10px">{safe_prompt}</textarea>
    <div style="margin-top:10px;display:flex;gap:10px">
      <button onclick="copyPromptFix({seq})" style="background:rgba(245,182,66,.15);border:1px solid rgba(245,182,66,.4);color:var(--gold);padding:8px 16px;border-radius:8px;cursor:pointer;font-size:13px;font-weight:700">📋 复制修改后的提示词</button>
      <button onclick="resetPrompt({seq})" style="background:rgba(255,92,138,.15);border:1px solid rgba(255,92,138,.4);color:var(--mag);padding:8px 16px;border-radius:8px;cursor:pointer;font-size:13px">↩️ 还原</button>
    </div>
    <p style="color:var(--mag);font-size:11px;margin-top:8px">⚠️ 提示词一经改动，sha1 即失效 → 必须重新过审（门禁会拦「审了 A 却跑了 B」）</p>
  </details>

    n_seg = len(reports)
    total_dur = sum(float(r.get("dur") or 0) for r in reports)

{resolution_html}

# ========== 07 成片审核（关 7） ==========

def build_final_review(project_root):
    project_root = Path(project_root)
    # V1.3（铁律 D20）：成片唯一权威路径是 <项目根>/视频/final/final.mp4。
    # 保留 <项目根>/final/final.mp4 作为老项目兜底（V1.3 之前 batch_run_drama.py
    # 的路径 bug 把成片写到了这里），但只在权威路径不存在时才用它。
    canonical = project_root / "视频" / "final" / "final.mp4"
    legacy = project_root / "final" / "final.mp4"
    if canonical.exists():
        final_mp4 = canonical
    elif legacy.exists():
        final_mp4 = legacy
    else:
        final_mp4 = None
    if final_mp4 is None:
        return None, "final.mp4 不存在（应位于 视频/final/final.mp4）"

    # 汇总所有段
    video_dir = project_root / "视频"
    rows = ""
    total_coins = 0
    total_secs = 0
    for shot_dir in sorted(video_dir.iterdir()):
        if not shot_dir.is_dir() or shot_dir.name == "final":
            continue
        log_file = shot_dir / "log.json"
        audit_file = shot_dir / "audit" / "audit_result.json"
        if not log_file.exists():
            continue
        log = json.loads(log_file.read_text(encoding="utf-8"))
        audit = json.loads(audit_file.read_text(encoding="utf-8")) if audit_file.exists() else {}

        total_coins += log.get("coins", 0)
        total_secs += log.get("secs", 0)
        overall = audit.get("overall", "未审")
        cls = "ok" if overall == "PASS" else ("fail" if overall == "FAIL" else "pending")
        face_cell = "—"
        audit_face = (audit or {}).get("face")
        if audit_face:
            if audit_face.get("applicable") is False:
                face_cell = "不适用"
            elif audit_face.get("pass") is True:
                face_cell = "人脸✅"
            elif audit_face.get("pass") is False:
                face_cell = "人脸❌"
            else:
                face_cell = "未判"

        rows += f"""<tr>
  <td>{html.escape(shot_dir.name)}</td>
  <td>{log.get('dur','?')}s</td>
  <td>{log.get('coins','?')}</td>
  <td>{log.get('secs','?')}s</td>
  <td class="{cls}">{overall}</td>
  <td>{face_cell}</td>
</tr>"""

    video_html = f'<video controls src="{_rel(final_mp4, project_root)}" style="max-width:960px;margin:20px auto;display:block"></video>'

    # 成片元信息（V1.3.1）：显式暴露文件时间，避免用户按资源管理器「创建日期」
    # 判断新旧——Windows 上原地覆盖不刷新创建时间，成片会显得「还是旧的」。
    _st = final_mp4.stat() if final_mp4.exists() else None
    _mtime = datetime.fromtimestamp(_st.st_mtime).strftime("%Y-%m-%d %H:%M:%S") if _st else "?"
    _size = ("%.1f MB" % (_st.st_size / 1048576.0)) if _st else "?"
    _dur = "?"
    try:
        _pr = subprocess.run(["ffprobe", "-v", "error", "-show_entries", "format=duration",
                              "-of", "default=nw=1:nk=1", str(final_mp4)],
                             capture_output=True, text=True, timeout=30)
        if _pr.returncode == 0 and _pr.stdout.strip():
            _dur = "%.2f s" % float(_pr.stdout.strip())
    except Exception:
        pass
    try:
        _disp = str(final_mp4.resolve().relative_to(project_root.resolve())).replace("\\", "/")
    except Exception:
        _disp = str(final_mp4)
    meta_html = (f'<p style="margin-top:14px;color:var(--mut);font-size:13px">'
                 f'<b style="color:var(--txt)">文件：</b>'
                 f'<code>{html.escape(_disp)}</code> · '
                 f'<b style="color:var(--txt)">大小：</b>{_size} · '
                 f'<b style="color:var(--txt)">时长：</b>{_dur} · '
                 f'<b style="color:var(--txt)">更新时间：</b>{_mtime}</p>')

    body = f"""
<div class="sect" style="text-align:center">
  <h2>🎬 成片</h2>
  {video_html}
  {meta_html}
</div>
<div class="sect">
  <h2>📊 分段汇总</h2>
  <table>
    <thead><tr><th>段</th><th>时长</th><th>成本(RH币)</th><th>耗时</th><th>总判决</th><th>人脸一致性</th></tr></thead>
    <tbody>{rows}</tbody>
  </table>
  <p style="margin-top:16px;color:var(--mut)">累计：<b style="color:var(--gold)">{total_coins} RH 币</b> · 生成耗时 <b>{total_secs}s</b></p>
</div>
{actions_block([
  "若整体通过，对话里说 <code>成片通过</code>，本剧交付完成",
  "若要重抽某段，说 <code>重抽段 N：xxx</code>（会自动重新拼接）",
  "若要看完整制作日志，打开 <code>视频/final/production_log.md</code>",
])}
"""
    out = project_root / "审核" / "07_成片审核.html"
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(page_shell("07 · 成片审核", project_root.name,
                              "最终交付", body,
                              _stage_state(project_root, "final")), encoding="utf-8")
    return out, None


# ========== 主入口 ==========

def main():
    ap = argparse.ArgumentParser(description="审核 HTML 生成器")
    ap.add_argument("--project-root", required=True)
    ap.add_argument("--stage", required=True,
                    choices=["script", "style", "assets", "storyboard", "prompt", "video", "final"])
    args = ap.parse_args()

    builders = {
        "script": build_script_review,
        "style": build_style_review,
        "assets": build_asset_review,
        "storyboard": build_storyboard_review,
        "prompt": build_prompt_review,
        "video": build_video_review,
        "final": build_final_review,
    }
    out, err = builders[args.stage](args.project_root)
    if err:
        sys.exit(f"❌ {err}")
    print(f"✅ {out}")


if __name__ == "__main__":
    main()
