def read_version():
    return (SKILL / "VERSION").read_text(encoding="utf-8").strip()


def count_invariants():
    return len([l for l in (TOOLS / "checks.txt").read_text(encoding="utf-8").splitlines()
                if l.strip() and not l.strip().startswith("#")])


def count_adversarial():
    return len([p for p in (SKILL / "examples").glob("test_*.py")])


def extract_rules():
    """从 SKILL.md 抽取铁律（code + 标题）。**抽取，不手抄。**"""
    skill = (SKILL / "SKILL.md").read_text(encoding="utf-8")
    pat = re.compile(r'^\*\*([A-G]\d+[a-z\-]*(?:\.\d+)?)\s*[\.·]?\s*(.{0,90})', re.M)
    rules, seen = [], set()
    for m in pat.finditer(skill):
        code = m.group(1).strip()
        title = re.sub(r'（.*?）', '', m.group(2)).strip('** ').strip()
        if title and code not in seen:
            seen.add(code)
            rules.append((code, title))
    return rules

def gen_history():
    """从 CHANGELOG.md 生成版本历史 HTML 行（最新 N 条）。"""
    cl = (SKILL / "CHANGELOG.md").read_text(encoding="utf-8")
    rows = []
    for m in re.finditer(r'^## V(\d+\.\d+(?:\.\d+)?)\s*·\s*(.*?)$', cl, re.M):
        ver, rest = m.group(1), m.group(2).strip()
        # 一句话摘要：取紧随其后的 `**一句话**：…`
        tail = cl[m.end():m.end() + 900]
        sm = re.search(r'\*\*一句话\*\*[：:]\s*(.+)', tail)
        gist = (sm.group(1).strip() if sm else rest)
        gist = re.split(r'\n', gist)[0].strip('*。 ')
        gist = re.sub(r'（[^）]*铁律[^）]*）', '', gist).strip()
        rows.append((ver, gist[:78]))
    rows = rows[:18]
    # ⚠ 用 <table> 而不是 <p> 列表：项目说明对表格的保留最稳，
    #   而 <p> 上的 display:inline-block / 负 margin 会被剥掉（铁律 D2-q）。
    html = ['  <table width="100%" cellspacing="0" cellpadding="0" border="0" '
            'style="border-collapse:collapse;">']
    for i, (ver, gist) in enumerate(rows):
        color = '#b3541e' if i == 0 else '#0f2b3d'
        html.append('    <tr>')
        html.append(f'      <td width="62" style="padding:6px 8px 6px 0;font-size:13px;'
                    f'line-height:1.75;font-weight:bold;color:{color};vertical-align:top;">V{ver}</td>')
        html.append(f'      <td style="padding:6px 0;font-size:13px;line-height:1.75;'
                    f'color:#4a4a4a;vertical-align:top;">{gist}</td>')
        html.append('    </tr>')
    html.append('  </table>')
    return "\n".join(html)


def fill_article(ver, n_rules, n_inv, n_adv):
    t = ARTICLE.read_text(encoding="utf-8")
    before = t
    # ⚠ 必须是**幂等**的：第一版用 `t.replace("<!--@VER@-->", ...)`，
    #   占位符**第一次填充后就被消耗掉**了 —— 第二次运行无处可填，
    #   于是「介绍页的版本徽章永远停在第一次填的那个版本」（实测：单文件版已是 V2.2.3，
    #   介绍页徽章还写着 V2.2.2）。**一次性占位符不是模板，是坑。**
    #   改为按「当前版本位」的上下文正则替换：token 在就用 token，不在就改现值。
    # ⚠ 三轮踩坑换来的写法（**别再按「当时那一段长什么样」写替换**）：
    #   ① 只认 `</b>` → 页面改成 `</span>` 后徽章**再也没被更新**（停在上一版）；
    #   ② 只认「恰好一个 V」→ 模板里已有字面 V 时出 `VV2.2.5`；
    #   ③ 只认 `.zip</code>` → 「怎么用」那处是 `</span>`，又漏了。
    #   ⇒ 现在只按**语义位**匹配，且**不依赖闭合标签**、容忍重复的 V（`V*`）：
    #      · 包名：`drama-auto-pipeline-V<版本>...`（历史表里没有包名，可全局替换）
    #      · 页头徽章 / 页脚 / 页面标题 / 版本历史副标题：各自唯一的前缀锚点
    _V = r'V*\d+\.\d+(?:\.\d+)?'
    _VER = 'V%s' % ver
    t = t.replace("<!--@VER@-->", _VER)                                  # 模板占位符（首次）
    t = re.sub(r'drama-auto-pipeline-' + _V, 'drama-auto-pipeline-' + _VER, t)   # 所有包名（含 <日期> 变体）
    t = re.sub(r'(font-size:15px;">)' + _V, r'\g<1>' + _VER, t)                 # 页头徽章
    t = re.sub(r'(color:#1d6f7a;">)' + _V, r'\g<1>' + _VER, t)                 # 页脚
    t = re.sub(r'(<title>drama-auto-pipeline )' + _V, r'\g<1>' + _VER, t)      # 页面标题
    t = re.sub(r'(从 V1\.0 到 )' + _V, r'\g<1>' + _VER, t)                      # 版本历史副标题
    t = t.replace("<!--@STATS@-->",
                  f'累计：<b style="color:#0f2b3d;">{n_rules} 条铁律</b> · '
                  f'<b style="color:#0f2b3d;">{n_inv} 条不变式</b> · '
                  f'<b style="color:#0f2b3d;">{n_adv} 个对抗测试</b> · '
                  f'<b style="color:#0f2b3d;">7 道审核关</b> · '
                  f'<b style="color:#0f2b3d;">4 道发版门禁</b>')
    a = t.find("<!--HISTORY-START-->")
    b = t.find("<!--HISTORY-END-->")
    if a >= 0 and b > a:
        head = t[:a + len("<!--HISTORY-START-->")]
        tail = t[b:]
        t = head + "\n" + gen_history() + "\n" + tail
    ARTICLE.write_text(t, encoding="utf-8")
    return t != before


def main():
    ap = argparse.ArgumentParser(description="生成/刷新交付物（单文件版 + 项目说明介绍页）")
    ap.add_argument("--check", action="store_true", help="只检查是否需要刷新，不写盘")
    args = ap.parse_args()

    ver = read_version()
    rules = extract_rules()
    n_inv, n_adv = count_invariants(), count_adversarial()

    st_new = gen_standalone()
    st_old = STANDALONE.read_text(encoding="utf-8") if STANDALONE.exists() else ""
    art_old = ARTICLE.read_text(encoding="utf-8") if ARTICLE.exists() else ""

    if args.check:
        need = []
        if st_new != st_old:
            need.append("drama-auto-pipeline-standalone.md")
        # 项目说明页：只在占位符还留着、或填出来与现状不同时需要刷新
        tmp = art_old
        for tok, val in (("<!--@VER@-->", f"V{ver}"),):
            if tok in art_old and val not in art_old:
                need.append("drama-auto-pipeline-项目说明介绍页.html")
                break
        if need:
            print("⚠ 交付物需要刷新：%s（跑 python tools/make_deliverables.py）" % "、".join(need))
            sys.exit(1)
        print("✅ 交付物与实现已同步（V%s · 铁律 %d · 不变式 %d · 对抗测试 %d）"
              % (ver, len(rules), n_inv, n_adv))
        sys.exit(0)

    STANDALONE.write_text(st_new, encoding="utf-8")
    changed_art = fill_article(ver, len(rules), n_inv, n_adv)

    # 3) 项目说明安全校验（铁律 D2-q）：**交付前必须过**，否则发版中断。
    #    判据：「在浏览器里好看」与「粘进项目说明还好看」是两件事 ——
    #    交付物要在它**真正要被使用的地方**验收，不能在「我这边看起来对」就收工。
    try:
        import subprocess as _sp2
        r = _sp2.run([sys.executable,
                      os.path.join(os.path.dirname(os.path.abspath(__file__)),
                                   "check_mp_html.py"), str(ARTICLE)],
                     capture_output=True, text=True, encoding="utf-8", errors="replace")
    except Exception as e:                                    # noqa: BLE001
        print("⚠️ 项目说明安全校验未能执行：%s: %s" % (type(e).__name__, e))
        r = None
    if r is not None:
        if r.returncode != 0:
            print(r.stdout or "")
            print("❌ 项目说明介绍页**没通过项目说明安全校验** —— 拒绝交付。")
            print("   （粘进项目说明会塌版面。排查：python tools/check_mp_html.py --all）")
            sys.exit(1)
        print("   ✅ 项目说明安全校验通过（粘贴即排版）")
    print("✅ 交付物已刷新（V%s）" % ver)
    print("   drama-auto-pipeline-standalone.md　%d 行 · 铁律 %d 条 · 通用判据 %d 条"
          % (st_new.count("\n") + 1, len(rules), len(JUS)))
    print("   drama-auto-pipeline-项目说明介绍页.html　%s"
          % ("已更新" if changed_art else "无变化"))
    print("   统计：不变式 %d 条 · 对抗测试 %d 个" % (n_inv, n_adv))
    sys.exit(0)


if __name__ == "__main__":
    main()
