#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""MV 音画对齐核验器（mv-auto-pipeline · 铁律 A5/B5/E4）

两个模式，对应两道关：

  --plan     关 6（跑视频之前的硬门禁）
             从 storyboard.json 汇总「音乐窗口计划」→ 状态/music_plan.json
             核：窗口连续无缝 / Σ窗口 = 全曲长 / 每段歌词行都落在自己窗口内 /
                 对口型段的景别是中近景 / 段长落 [4,15]
             batch_run_mv.py 会读它，不过就直接拒绝生成（零算力）

  --verify   关 8（成片之后的对齐三验）
             ① 时长：Σ各段 output.mp4 == 原曲长（±0.15s）
             ② 波形：逐段提取 RMS 包络，在「预期位置 ±1.5s」内做局部相关 ——
                最优滞后量应 ≈ 0s（这就是「音乐对没对上」的客观证据）
             ③ 人声：对口型段的能量必须来自母带人声（能量非空 + 与母带同位置相关）
             输出 视频/final/对齐验收.html

为什么用「包络 + 局部搜索」而不是原始互相关：
  213 秒的歌按 8k 单声道起算接近 1.7M 采样，纯 Python 全量互相关是十亿级运算。
  先降到 10 帧/秒的 RMS 包络（2 千点），再只在预期位置附近搜 ±1.5s，
  运算量降到万级 —— 纯标准库就能秒级出结果，且判据完全够用。

用法：
  python check_alignment.py --project-root "<项目根>" --plan
  python check_alignment.py --project-root "<项目根>" --verify
"""
import argparse
import array
import html
import json
import math
import os
import re
import shutil
import subprocess
import sys

FFMPEG = next((p for p in ("ffmpeg", r"D:\AIxiufu\ffmpeg.exe", r"C:\ffmpeg\bin\ffmpeg.exe")
               if (os.path.isfile(p) or shutil.which(p))), "ffmpeg")
FFPROBE = next((p for p in ("ffprobe", r"D:\AIxiufu\ffprobe.exe", r"C:\ffmpeg\bin\ffprobe.exe")
                if (os.path.isfile(p) or shutil.which(p))), "ffprobe")
SR = 8000
HOP = 800          # 0.1s 一帧包络
SYNC_SHOTS = {"ECU", "CU", "MCU", "MS"}


def read_json(p, d=None):
    try:
        return json.load(open(p, encoding="utf-8"))
    except (OSError, ValueError):
        return d


def write_json(p, o):
    os.makedirs(os.path.dirname(os.path.abspath(p)) or ".", exist_ok=True)
    tmp = p + ".tmp"
    json.dump(o, open(tmp, "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    os.replace(tmp, p)


def find_song(root):
    md = os.path.join(root, "音乐")
    for f in ("song.mp3", "song.wav", "song.flac", "song.m4a", "song.aac"):
        if os.path.isfile(os.path.join(md, f)):
            return os.path.join(md, f)
    if os.path.isdir(md):
        for f in sorted(os.listdir(md)):
            if f.lower().endswith((".mp3", ".wav", ".flac", ".m4a", ".aac")) and not f.startswith("cut_"):
                return os.path.join(md, f)
    return None


def dur_of(p):
    try:
        r = subprocess.run([FFPROBE, "-v", "error", "-show_entries", "format=duration",
                            "-of", "default=nw=1:nk=1", p], capture_output=True, text=True, timeout=120)
        return round(float((r.stdout or "0").strip() or 0), 3)
    except Exception:  # noqa: BLE001
        return None


def pcm(path, ss=None, t=None):
    """取 8k 单声道 s16le PCM → array('h')。"""
    cmd = [FFMPEG, "-v", "error"]
    if ss is not None:
        cmd += ["-ss", f"{ss:.3f}"]
    cmd += ["-i", path]
    if t is not None:
        cmd += ["-t", f"{t:.3f}"]
    cmd += ["-vn", "-ac", "1", "-ar", str(SR), "-f", "s16le", "-"]
    r = subprocess.run(cmd, capture_output=True, timeout=1800)
    a = array.array("h")
    a.frombytes(r.stdout[:len(r.stdout) // 2 * 2])
    return a


def envelope(samples, hop=HOP):
    """RMS 包络（10 帧/秒）。"""
    env = []
    for i in range(0, len(samples) - hop, hop):
        s = 0
        for v in samples[i:i + hop]:
            s += v * v
        env.append(math.sqrt(s / hop))
    return env


def corr(a, b):
    """归一化相关（不用 numpy）。"""
    n = min(len(a), len(b))
    if n < 4:
        return 0.0
    a, b = a[:n], b[:n]
    ma, mb = sum(a) / n, sum(b) / n
    num = sum((x - ma) * (y - mb) for x, y in zip(a, b))
    da = math.sqrt(sum((x - ma) ** 2 for x in a))
    db = math.sqrt(sum((y - mb) ** 2 for y in b))
    if da < 1e-6 or db < 1e-6:
        return 0.0
    return num / (da * db)


# ------------------------------------------------------------------ 关 6：计划核对
def do_plan(root, sb):
    song = find_song(root)
    total = (sb.get("song") or {}).get("duration") or (dur_of(song) if song else None)
    problems, warns = [], []
    windows, cursor = {}, 0.0
    for seg in sb["segments"]:
        w = seg["music_window"]
        windows[str(seg["seq"])] = {"start": w["start"], "end": w["end"], "dur": w["dur"],
                                    "sing_mode": seg["sing_mode"],
                                    "shot_size": seg["shot_size"],
                                    "lyric_lines": [l["idx"] for l in seg["lyric_lines"]
                                                    if l["kind"] == "lyric"]}
        if abs(w["start"] - cursor) > 0.05:
            problems.append(f"段 {seg['seq']} 窗口起点 {w['start']}s ≠ 上一段终点 {cursor}s")
        if w["dur"] < 4.0 - 0.05 or w["dur"] > 15.05:
            problems.append(f"段 {seg['seq']} 窗口长 {w['dur']}s 越出 [4,15]")
        if seg["sing_mode"] == "sync" and seg["shot_size"] not in SYNC_SHOTS:
            problems.append(f"段 {seg['seq']} 对口型但景别 {seg['shot_size']} 不是中近景"
                            f"（铁律 B1）")
        for l in seg["lyric_lines"]:
            if l["kind"] != "lyric":
                continue
            if l["start"] < w["start"] - 0.05 or l["end"] > w["end"] + 0.35:
                problems.append(f"段 {seg['seq']} 的歌词行 {l['idx']}（{l['start']}–{l['end']}s）"
                                f"越出窗口 {w['start']}–{w['end']}s")
        cursor = w["end"]
    if total:
        if abs(cursor - total) > 0.5:
            problems.append(f"末段窗口终点 {cursor:.2f}s 与歌曲长 {total:.2f}s 差 "
                            f"{abs(cursor-total):.2f}s")
        else:
            warns.append(f"窗口合计 {cursor:.2f}s ≈ 曲长 {total:.2f}s（差 {abs(cursor-total):.3f}s）")
    syncs = [s for s in sb["segments"] if s["sing_mode"] == "sync"]
    ratio = len(syncs) / max(1, len(sb["segments"]))
    if not (0.30 <= ratio <= 0.65):
        warns.append(f"对口型段占比 {ratio:.0%} 在建议区间 35%~60% 之外 —— 确认这是你要的")
    run = mx = 0
    for s in sb["segments"]:
        run = run + 1 if s["sing_mode"] == "sync" else 0
        mx = max(mx, run)
    if mx > 3:
        warns.append(f"连续对口型段最多 {mx} 段（建议 ≤3，否则观众会腻）")

    plan = {"kind": "mv", "checked_at": __import__("datetime").datetime.now().isoformat(timespec="seconds"),
            "song_duration": total, "segments": len(sb["segments"]),
            "window_sum": round(cursor, 3), "sync_ratio": round(ratio, 3),
            "ok": not problems, "problems": problems, "warns": warns, "windows": windows}
    write_json(os.path.join(root, "状态", "music_plan.json"), plan)

    print(f"[关 6] 音乐窗口计划：{len(sb['segments'])} 段 · 窗口合计 {cursor:.2f}s / 曲长 {total}s "
          f"· 对口型 {ratio:.0%}")
    for w in warns:
        print("  ⚠", w)
    if problems:
        print(f"\n⛔ {len(problems)} 项不过（batch_run_mv.py 会拒绝生成）：")
        for p in problems:
            print("   ·", p)
    else:
        print("✅ 关 6 全过（窗口连续 / 段长合法 / 口型-景别一致 / 歌词不越窗）")
    return plan


# ------------------------------------------------------------------ 关 8：成片三验
def do_verify(root, sb, max_lag=1.5):
    song = find_song(root)
    if not song:
        sys.exit("❌ 找不到原曲文件（音乐/song.mp3）")
    total = (sb.get("song") or {}).get("duration") or dur_of(song)
    song_env = envelope(pcm(song))
    print(f"[song] 包络 {len(song_env)} 帧（0.1s/帧）· 曲长 {total}s")

    rows, hard = [], []
    sum_dur = 0.0
    for seg in sb["segments"]:
        kw = re.sub(r"[\s/\\:*?\"<>|]", "", seg.get("kw") or "seg")[:10] or "seg"
        p = os.path.join(root, "视频", f"shot_{seg['seq']:02d}_{kw}", "output.mp4")
        if not os.path.isfile(p):
            alt = [d for d in os.listdir(os.path.join(root, "视频"))
                   if d.startswith(f"shot_{seg['seq']:02d}_")]
            p = os.path.join(root, "视频", alt[0], "output.mp4") if alt else p
        if not os.path.isfile(p):
            hard.append(f"段 {seg['seq']} 找不到 output.mp4")
            continue
        d = dur_of(p) or 0
        sum_dur += d
        w = seg["music_window"]
        lag, c, rms_seg = None, None, None
        try:
            seg_env = envelope(pcm(p))
            rms_seg = round(sum(seg_env) / max(1, len(seg_env)), 1)
            exp = int(w["start"] * 10)
            best = (0.0, -2.0)
            step = 1
            for off in range(-int(max_lag * 10), int(max_lag * 10) + 1, step):
                i0 = exp + off
                # ★ V1.0.2 修（**校验器假阳性**，《野心家》关 8 实测暴露）：
                #   原写 `i0 = max(0, exp + off)` —— 段 1 的 exp = 0，于是 off = -15..-1
                #   全被夹到 0，等于**同一段参考被反复比较**；而取「首个最优」（严格 >）
                #   时 off = -15 先中选 → 滞后被误报成 **-1.5s**（相关系数明明是 1.00，
                #   一点都不像对不齐）。判据：**相关度满分却报大滞后 = 校验器的问题**。
                #   正解：越界的偏移**直接跳过**，不 clamp。
                if i0 < 0:
                    continue
                ref = song_env[i0:i0 + len(seg_env)]
                if len(ref) < len(seg_env) * 0.8:
                    continue
                cc = corr(seg_env, ref)
                if cc > best[1]:
                    best = (off / 10.0, cc)
            lag, c = best
        except Exception as e:  # noqa: BLE001
            hard.append(f"段 {seg['seq']} 音频分析失败：{type(e).__name__}")
        flag = []
        if d and abs(d - w["dur"]) > 0.12:
            flag.append(f"时长 {d:.2f}s ≠ 窗口 {w['dur']:.2f}s")
        if lag is not None and abs(lag) > 0.35:
            flag.append(f"音画滞后 {lag:+.1f}s")
        if c is not None and c < 0.35:
            flag.append(f"波形相关度低（{c:.2f}）")
        if seg["sing_mode"] == "sync" and rms_seg is not None and rms_seg < 60:
            flag.append(f"对口型段音频能量过低（RMS {rms_seg}）")
        rows.append({"seq": seg["seq"], "dur": d, "win": w["dur"], "lag": lag, "corr": c,
                     "rms": rms_seg, "sing": seg["sing_mode"], "shot": seg["shot_size"],
                     "start": w["start"], "flags": flag, "file": os.path.relpath(p, root)})
        if flag:
            hard.append(f"段 {seg['seq']}：" + "；".join(flag))

    tail = ""
    if total and abs(sum_dur - total) > 0.4:
        tail = f"各段合计 {sum_dur:.2f}s 与曲长 {total:.2f}s 差 {abs(sum_dur-total):.2f}s"
        hard.append(tail)

    print(f"\n[关 8] 对齐三验：{len(rows)} 段 · 各段合计 {sum_dur:.2f}s / 曲长 {total}s")
    for r in rows:
        tag = "✅" if not r["flags"] else "⛔"
        lag_s = "—" if r["lag"] is None else f"{r['lag']:+.1f}s"
        corr_s = "—" if r["corr"] is None else f"{r['corr']:.2f}"
        print(f"  {tag} 段 {r['seq']:>2} 时长 {r['dur']:>6.2f}/{r['win']:>6.2f}s "
              f"滞后 {lag_s:>7} 相关 {corr_s:>5} {r['sing']} {r['shot']}"
              + ("  " + "；".join(r["flags"]) if r["flags"] else ""))
    if hard:
        print(f"\n⛔ {len(hard)} 项未通过：")
        for h in hard:
            print("   ·", h)
    else:
        print("\n✅ 对齐三验全过：段长贴合 / 音画滞后 ≈ 0 / 对口型段能量正常")

    out = os.path.join(root, "视频", "final", "对齐验收.html")
    os.makedirs(os.path.dirname(out), exist_ok=True)
    trlist = []
    for r in rows:
        lag_s = "—" if r["lag"] is None else f"{r['lag']:+.1f}s"
        corr_s = "—" if r["corr"] is None else f"{r['corr']:.2f}"
        trlist.append(
            f"<tr class='{'bad' if r['flags'] else ''}'><td>{r['seq']}</td>"
            f"<td>{r['start']:.2f}s</td><td>{r['win']:.2f}s</td><td>{r['dur']:.2f}s</td>"
            f"<td>{lag_s}</td><td>{corr_s}</td><td>{r['rms']}</td>"
            f"<td>{r['sing']}</td><td>{r['shot']}</td>"
            f"<td>{html.escape('；'.join(r['flags']) or '✅')}</td></tr>")
    tr = "".join(trlist)
    doc = f"""<!DOCTYPE html><html lang="zh-CN"><head><meta charset="utf-8">
<title>MV 对齐验收</title><style>
body{{margin:0;background:#f6f7f9;color:#1c2025;font:14px/1.6 -apple-system,"Segoe UI","Microsoft YaHei",sans-serif}}
header{{background:#fff;border-bottom:1px solid #e3e6ea;padding:18px 24px}}
h1{{margin:0 0 6px;font-size:18px}}.kpi{{display:flex;gap:18px;color:#5b6570;font-size:13px;flex-wrap:wrap}}
main{{padding:20px 24px 60px}}.wrap{{background:#fff;border:1px solid #e3e6ea;border-radius:10px;overflow:auto}}
table{{width:100%;border-collapse:collapse;font-size:13px}}
th,td{{padding:7px 10px;border-bottom:1px solid #e3e6ea;text-align:left}}
th{{background:#fafbfc;color:#5b6570;position:sticky;top:0}}
tr.bad{{background:#fdecea}}</style></head><body>
<header><h1>MV 对齐验收（关 8）</h1><div class="kpi">
<span>段数 <b>{len(rows)}</b></span><span>各段合计 <b>{sum_dur:.2f}s</b></span>
<span>曲长 <b>{total}s</b></span>
<span>结论 <b>{'⛔ 有问题' if hard else '✅ 全过'}</b></span></div></header>
<main><div class="wrap"><table><thead><tr>
<th>段</th><th>窗口起点</th><th>窗口长</th><th>视频长</th><th>音画滞后</th>
<th>波形相关</th><th>RMS</th><th>口型</th><th>景别</th><th>结论</th></tr></thead>
<tbody>{tr}</tbody></table></div></main></body></html>"""
    open(out, "w", encoding="utf-8").write(doc)
    print(f"\n验收页：{out}")
    return rows, hard


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--project-root", required=True)
    ap.add_argument("--plan", action="store_true")
    ap.add_argument("--verify", action="store_true")
    ap.add_argument("--max-lag", type=float, default=1.5)
    args = ap.parse_args()
    root = os.path.abspath(args.project_root)
    sb = read_json(os.path.join(root, "分镜脚本", "storyboard.json"))
    if not sb:
        sys.exit("❌ 读不到 分镜脚本/storyboard.json")
    if args.plan or not args.verify:
        do_plan(root, sb)
    if args.verify:
        do_verify(root, sb, args.max_lag)


if __name__ == "__main__":
    main()
