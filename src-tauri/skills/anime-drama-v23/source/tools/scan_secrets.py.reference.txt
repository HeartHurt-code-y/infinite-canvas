#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""drama-auto-pipeline 打包前**凭证泄漏**扫描器

================================================================================
为什么需要它（和 scan_paths.py 是一对）
================================================================================
`scan_paths.py` 拦的是「写死了作者机器的路径」；
本脚本拦的是「把作者的真实密钥打进了分发包」。

两者的判据是同一个：**「碰巧没事」≠「不会出事」。**
包里现在是干净的，只说明「恰好那个文件不在」，**不说明有东西在拦**。
所以不靠「我记得检查」，靠这个脚本 + 接进 snapshot.py 发版流程硬拦。

**凭证是第三条出口。** 打包有两条出口（`_versions/*.zip` 给别人用、git 版本库），
路径靠扫描器 + `.gitignore` 各堵一条；密钥则**两条出口都可能漏** —— 本脚本同时扫
工作区与包本体。

================================================================================
三条设计纪律
================================================================================
① **不硬编码任何真实密钥**。指纹在**运行时**从用户级凭证目录读
   （`~/.项目模型/credentials/**/keys.env`），只用于比对。
   否则扫描器自己就成了泄漏源 —— 这是 mv-auto-pipeline 那边实测踩过的自指坑。
② **不回显密钥本体**。命中只打印掩码（`sk-abc***xyz`）+ 文件 + 行号。
   报告本身也是可能被转发的东西。
③ **对抗测试必须主动伪造泄漏**。只跑一次「现在干净」等于没测防护。
   见 `examples/test_scan_secrets.py`（假密钥**运行时生成**，不写进源码）。

================================================================================
不报什么（误报也是故障）
================================================================================
- **URL / 端点**（`[项目内资源]`）：是公开地址，不是密钥
- **示例占位符**（`sk-xxxx`、`<你的key>`、`YOUR_project_provider_setting`、`%项目模型_project_provider_setting%`）
- **纯 sha256 / 内容哈希文件名**：`vendor/MANIFEST.json` 的校验值、
  ComfyUI 工作流里 `"audio": "<64位hex>.wav"` 这种上传资源文件名
  —— 早期把这两类报成泄漏，一次 156 条噪声，报告直接没人看

用法：
    python tools/scan_secrets.py                 # 扫 skill 目录（默认）
    python tools/scan_secrets.py --zip _versions/xxx.zip
    python tools/scan_secrets.py --all-zips      # 复检所有历史包

退出码：0 = 干净；3 = 命中（**不许发版**）
"""
import argparse
import os
import re
import sys
import zipfile
from pathlib import Path

try:
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")
except Exception:                                    # noqa: BLE001
    pass

SKILL_DIR = Path(__file__).resolve().parent.parent
SEP = chr(92)

SCAN_EXT = {".py", ".md", ".txt", ".json", ".jsonc", ".html", ".htm", ".css", ".js",
            ".yaml", ".yml", ".toml", ".ini", ".cfg", ".sh", ".ps1", ".bat", ".cmd",
            ".env", ".example", ".template", ".csv", ".tsv", ".xml", ".log"}
SKIP_DIRS = {".git", "__pycache__", "_versions", "node_modules", ".venv", "venv",
             ".mypy_cache", ".pytest_cache", ".idea", ".vscode"}
MAX_BYTES = 4 * 1024 * 1024

# 凭证文件本身：出现在包里就是问题（不管内容是什么）
CRED_FILENAMES = {"keys.env", ".env", "token.json", "secrets.json", "credentials.json"}
CRED_EXTS = {".key", ".pem", ".p12", ".pfx", ".keystore", ".jks", ".ppk"}

# 有辨识度的密钥前缀（挡「别的平台」的密钥，不依赖本机凭证文件）
KNOWN_PREFIX = [
    ("OpenAI/兼容 sk- 密钥", re.compile(r"sk-[A-Za-z0-9_\-]{16,}")),
    ("项目模型 rh- 密钥", re.compile(r"rh-[A-Za-z0-9_\-]{16,}")),
    ("AWS AccessKey", re.compile(r"\bAKIA[0-9A-Z]{16}\b")),
    ("HTTP Bearer 令牌", re.compile(r"Bearer\s+[A-Za-z0-9._\-]{20,}")),
    ("私钥块", re.compile(r"-----BEGIN [A-Z ]*PRIVATE KEY-----")),
]

# `project_provider_setting = "..."` / `"project_provider_setting": "..."` 这类字段：值够长就当疑似密钥。
# 分成「带引号」和「不带引号」两条，因为不带引号的一大半是**代码**不是值
# （实测：`project_provider_setting = get_project_provider_setting(args.project_provider_setting)` 被报成泄漏 → 见 _looks_like_code_value）。
KEYISH_NAMES = (r"api[_-]?key|project_provider_setting|secret[_-]?key|access[_-]?key|"
                r"auth[_-]?token|access[_-]?token|client[_-]?secret|"
                r"private[_-]?key|password|passwd|token")
KEYISH_FIELD_Q = re.compile(
    r"""(?ix) ["']?\b(""" + KEYISH_NAMES + r""")\b["']? \s*[:=]\s* ["']([^"'\n]{16,160})["'] """)
KEYISH_FIELD_U = re.compile(
    r"""(?ix) ["']?\b(""" + KEYISH_NAMES + r""")\b["']? \s*[:=]\s* """
    r"""([A-Za-z0-9_\-\.\+/=:~]{20,160})""")

# 明显是占位符 / 模板的值 —— 这些是**要教给用户**的写法，不是泄漏
PLACEHOLDER_MARKS = ("<", ">", "%", "{", "}", "你的", "在此", "填", "xxx", "XXX",
                     "YOUR", "your", "here", "TODO", "CHANGE", "REPLACE", "…")
# 纯哈希：32~128 位 hex —— **只在通用规则里**当作「校验值/内容哈希文件名」放行。
# ⚠️ 绝不能用它过滤「本机真实凭证」：项目模型 的密钥本身就是 32 位 hex，
#    拿它过滤等于把最该比对的那个值排除掉（实测踩到：只比对到 1 个本机凭证值）。
PURE_HEX = re.compile(r"^[0-9a-fA-F]{32,128}$")
HEX_FILENAME = re.compile(r"[0-9a-fA-F]{32,128}\.[A-Za-z0-9]{2,5}$")


def _looks_like_url(v: str) -> bool:
    """端点/URL 是公开地址，不是密钥。

    实测踩到：凭证文件里 `项目模型_API_BASE=[项目内资源]` 被当成密钥，
    于是代码里**合法**写着这个端点的那一行被报成泄漏 —— 一次误报就能让人开始跳过报告。
    """
    return "://" in v or v.lower().startswith(("http", "https", "ws", "wss"))


def _is_placeholder(v: str) -> bool:
    """通用规则用：值是不是「明显不是密钥」的模板/哈希。"""
    if not v or len(v) < 12:
        return True
    if any(m in v for m in PLACEHOLDER_MARKS):
        return True
    if set(v.lower()) <= set("x*.-_"):               # sk-xxxx 这种
        return True
    if PURE_HEX.match(v) or HEX_FILENAME.search(v):   # sha256 / 内容哈希文件名
        return True
    return False


def _obvious_placeholder(v: str) -> bool:
    """**本机凭证指纹**用：只挡「一眼就不是密钥」的值。

    和 `_is_placeholder` 的区别只有一条但很关键：**这里不挡纯 hex**。
    本机 项目模型 密钥就是 32 位 hex —— 用它挡等于不比对真密钥，防护形同虚设。
    """
    if len(v) < 16:
        return True
    if any(m in v for m in ("<", ">", "%", "{", "}", "你的", "在此", "填")):
        return True
    return set(v.lower()) <= set("x*.-_")


def _looks_like_code_value(v: str) -> bool:
    """`project_provider_setting = get_project_provider_setting(args.project_provider_setting)` 这种「等号右边是代码」的情况。

    实测踩到：这条把 `项目模型_image_runner.py:233` 报成「疑似密钥字段」，
    而那一行只是把命令行参数交给 helper。误报的代价是用户开始跳过整份报告。
    """
    if re.fullmatch(r"[A-Za-z_][A-Za-z0-9_]*(\.[A-Za-z_][A-Za-z0-9_]*)+", v):
        return True                                   # 属性路径 args.project_provider_setting
    if re.fullmatch(r"[A-Z][A-Z0-9_]{3,}", v):
        return True                                   # 变量名 项目模型_project_provider_setting
    if re.fullmatch(r"[A-Za-z_][A-Za-z0-9_]*", v) and not re.search(r"\d", v):
        return True                                   # 纯标识符（含下划线，无数字）
    if v.startswith(("get_", "os.", "sys.", "self.")) or "(" in v:
        return True                                   # 函数调用
    return False


def real_secrets() -> dict:
    """运行时收集本机真实凭证值（**绝不硬编码**）。

    只认「像密钥」的值：长度 ≥ 16 且不是 URL、不是占位符。
    输出永远不会被打印本体 —— 调用方只用 `mask()` 显示。
    """
    out = {}
    root = Path.home() / ".项目模型" / "credentials"
    if not root.is_dir():
        return out
    for p in root.rglob("*"):
        if not p.is_file():
            continue
        if p.name != "keys.env" and p.suffix.lower() not in (".env", ".txt"):
            continue
        try:
            text = p.read_text(encoding="utf-8", errors="replace")
        except OSError:
            continue
        for line in text.splitlines():
            line = line.strip()
            if not line or line.startswith("#") or "=" not in line:
                continue
            k, v = line.split("=", 1)
            v = v.strip().strip('"').strip("'")
            if not _looks_like_url(v) and not _obvious_placeholder(v):
                out[k.strip()] = v
    return out


def mask(s: str) -> str:
    """只给掩码 —— 报告也可能被转发。"""
    if len(s) <= 8:
        return "*" * len(s)
    return s[:4] + "*" * 6 + s[-4:]


def _scan_text(text, rel, secrets):
    hits = []
    lines = text.splitlines()

    def add(line_no, kind, masked, line, value=None):
        hits.append({"rel": rel, "line": line_no, "kind": kind, "masked": masked,
                     "snippet": _redact(line.strip()[:140], secrets, value)})

    for i, line in enumerate(lines, 1):
        # ① 本机真实凭证值（最硬的一条）
        for name, val in secrets.items():
            if val in line:
                add(i, "本机真实凭证 %s" % name, mask(val), line, val)
                break
        else:
            # ② 有辨识度的密钥前缀
            hit = False
            for kind, rx in KNOWN_PREFIX:
                for m in rx.finditer(line):
                    if _is_placeholder(m.group(0)) or _looks_like_url(m.group(0)):
                        continue
                    add(i, kind, mask(m.group(0)), line, m.group(0))
                    hit = True
                    break
                if hit:
                    break
            # ③ `project_provider_setting = "长串"` 这类字段（先看带引号的，可靠得多）
            #    ⚠️ 这里用 `_obvious_placeholder` 而**不是** `_is_placeholder`：
            #       字段名已经声明了「这是密钥」，所以值哪怕是纯 hex 也要报
            #       （实测：`{"project_provider_setting": "<40位hex>"}` 被「纯 hex = 哈希」放过了）。
            #       反过来说，`"audio": "<64位hex>.wav"` 这种因为字段名不 key-ish，
            #       压根进不到这两条规则，不会被误报。
            if not hit:
                m = KEYISH_FIELD_Q.search(line)
                if m:
                    val = m.group(2).strip()
                    if not (_obvious_placeholder(val) or _looks_like_url(val)
                            or _looks_like_code_value(val)):
                        add(i, "疑似密钥字段 %s" % m.group(1), mask(val), line, val)
                        hit = True
            # 只有这一条不带引号（`project_provider_setting=abcd1234...`），再要求「含数字」
            if not hit:
                m = KEYISH_FIELD_U.search(line)
                if m:
                    val = m.group(2).strip()
                    if not (_obvious_placeholder(val) or _looks_like_url(val)
                            or _looks_like_code_value(val)) and re.search(r"\d", val):
                        add(i, "疑似密钥字段 %s（无引号）" % m.group(1), mask(val), line, val)
    return hits


def _redact(text, secrets, value=None):
    """把行内出现过的密钥打成码 —— 报告本身也不能带真密钥。

    `value` 是本次命中的那个串（通用规则命中的也要打码，不只本机凭证）：
    报告可能被转发给第三方看，把密钥原样印在报告里等于又泄漏一次。
    """
    for _name, val in secrets.items():
        if val in text:
            text = text.replace(val, mask(val))
    if value and value in text:
        text = text.replace(value, mask(value))
    return text


def _is_cred_file(fn: str) -> bool:
    return fn in CRED_FILENAMES or Path(fn).suffix.lower() in CRED_EXTS


def _iter_files(root: Path):
    for r, dirs, files in os.walk(root):
        dirs[:] = [d for d in dirs if d not in SKIP_DIRS]
        for fn in files:
            p = Path(r) / fn
            # ⚠️ 凭证文件**必须**放行到这里 —— 它的后缀不在 SCAN_EXT 里（.pem/.key），
            #    早期这一层就把它们 continue 掉了，于是「凭证文件本体」这条规则从来没生效过
            #    （实测：`certs/server.pem` 实得 0）。
            if p.suffix.lower() not in SCAN_EXT and not _is_cred_file(fn):
                continue
            try:
                if p.stat().st_size > MAX_BYTES:
                    continue
            except OSError:
                continue
            yield p, str(p.relative_to(root)).replace(os.sep, "/")


def _cred_file_hit(rel: str):
    name = os.path.basename(rel)
    if _is_cred_file(name):
        return {"rel": rel, "line": 0, "kind": "凭证文件本体（不该进包）",
                "masked": "", "snippet": rel}
    return None


def scan_tree(root, quiet=False):
    root = Path(root)
    secrets = real_secrets()
    hits, n = [], 0
    for p, rel in _iter_files(root):
        n += 1
        cf = _cred_file_hit(rel)
        if cf:
            hits.append(cf)
            continue
        try:
            hits += _scan_text(p.read_text(encoding="utf-8", errors="replace"), rel, secrets)
        except OSError:
            continue
    return {"target": str(root), "kind": "dir", "files": n, "hits": hits,
            "n_secrets": len(secrets)}


def scan_zip(zpath, quiet=False):
    zpath = Path(zpath)
    secrets = real_secrets()
    hits, n = [], 0
    with zipfile.ZipFile(zpath) as z:
        for info in z.infolist():
            if info.is_dir():
                continue
            cf = _cred_file_hit(info.filename)
            if cf:
                hits.append(cf)
                continue
            if Path(info.filename).suffix.lower() not in SCAN_EXT or info.file_size > MAX_BYTES:
                continue
            n += 1
            try:
                hits += _scan_text(z.read(info).decode("utf-8", "replace"),
                                   info.filename, secrets)
            except Exception:                        # noqa: BLE001
                continue
    return {"target": str(zpath), "kind": "zip", "files": n, "hits": hits,
            "n_secrets": len(secrets)}


def report(res, quiet=False):
    hits = res["hits"]
    ok = not hits
    if not quiet or not ok:
        print("=" * 78)
        print("凭证泄漏扫描：%s（%s，%d 个文本文件；运行时比对了 %d 个本机凭证值）"
              % (res["target"], res["kind"], res["files"], res.get("n_secrets", 0)))
        print("=" * 78)
        if hits:
            print(f"\n❌ 命中 {len(hits)} 处 —— **不许发版**（收到包的人就拿到了你的密钥）：")
            for h in hits[:40]:
                loc = f"{h['rel']}:{h['line']}" if h["line"] else h["rel"]
                print(f"   {loc}  [{h['kind']}]  {h['masked']}  {h['snippet']}")
            if len(hits) > 40:
                print(f"   …… 另有 {len(hits) - 40} 处")
            print("\n   修法：从代码里删掉明文密钥，改走环境变量或凭证文件；")
            print("         并**重新打一个干净的包**（旧包已泄漏，光改工作区不够）。")
        else:
            print("\n✅ 干净：没有真实凭证、没有密钥指纹、没有凭证文件（命中值一律打码）")
    return ok


def main():
    ap = argparse.ArgumentParser(description="drama-auto-pipeline 凭证泄漏扫描器")
    ap.add_argument("--path", help="扫描目录（默认：本 skill 根目录）")
    ap.add_argument("--zip", help="扫描指定 zip")
    ap.add_argument("--all-zips", action="store_true", help="复检 _versions/ 下所有历史包")
    ap.add_argument("--quiet", action="store_true", help="只在有问题时输出")
    args = ap.parse_args()

    if args.all_zips:
        vd = SKILL_DIR / "_versions"
        zips = sorted(vd.glob("*.zip")) if vd.is_dir() else []
        if not zips:
            print("（_versions/ 下没有 zip）")
            return 0
        bad = 0
        for z in zips:
            res = scan_zip(z)
            flag = "❌ 命中 %d" % len(res["hits"]) if res["hits"] else "✅ 干净"
            print(f"  {flag:<12} {z.name}（{res['files']} 文件）")
            if res["hits"]:
                bad += 1
                for h in res["hits"][:6]:
                    print(f"        {h['rel']}:{h['line']}  [{h['kind']}]  {h['snippet']}")
        print(f"\n复检 {len(zips)} 个包：{len(zips) - bad} 干净 / {bad} 命中")
        return 3 if bad else 0

    res = scan_zip(args.zip) if args.zip else scan_tree(args.path or SKILL_DIR)
    ok = report(res, quiet=args.quiet)
    return 0 if ok else 3


if __name__ == "__main__":
    sys.exit(main())
