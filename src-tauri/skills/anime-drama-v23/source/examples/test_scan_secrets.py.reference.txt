#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""tools/scan_secrets.py 的对抗测试 —— **主动伪造泄漏场景，验证它真的会拦住**

================================================================================
为什么必须有这个文件
================================================================================
只跑一次「现在扫出来是干净的」**等于没测防护** ——
它只能证明「此刻恰好没有」，证明不了「以后出现了会被拦住」。

另外它还要防**另一个方向**的退化：**误报**。
一个门禁只要经常报假警，人就会整段跳过它，那它等于不存在。
（本脚本的样本里就固定留着四条曾经的真实误报：端点 URL、`sk-xxxx` 占位、
`project_provider_setting = get_project_provider_setting(...)` 代码行、sha256/内容哈希文件名。）

================================================================================
自指坑（这边尤其要紧）
================================================================================
假密钥**运行时生成**，源码里不出现任何密钥字面量 ——
否则「测试扫描器」的文件自己就成了扫描器的命中项，
而给扫描器加白名单放过测试文件 = 削弱防护（不干）。

用法：python examples/test_scan_secrets.py
退出码：0 = 全过
"""
import hashlib
import os
import shutil
import sys
import tempfile
import zipfile
from pathlib import Path

SKILL = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(SKILL / "tools"))
import scan_secrets as ss                                  # noqa: E402

# ---------------------------------------------------------------- 运行时生成的假密钥
# 32 位 hex：**刻意模仿 项目模型 的真实密钥形状** —— 用于回归「纯 hex 也算密钥」这条。
FAKE_HEX = hashlib.sha256(b"drama-test-1").hexdigest()[:32]
FAKE_SK = "sk-" + hashlib.sha256(b"drama-test-2").hexdigest()[:28]
FAKE_RH = "rh-" + hashlib.sha256(b"drama-test-3").hexdigest()[:24]
FAKE_BEARER = "Bearer " + hashlib.sha256(b"drama-test-4").hexdigest()[:32]
FAKE_JSON = hashlib.sha256(b"drama-test-5").hexdigest()[:40]
# 通用规则用的样本也要**运行时拼**：本文件自己会被「扫描器不报自己」那条断言扫到，
# 字面量写进源码就会自指命中（这和给扫描器加白名单不一样 —— 这里只是不写死样本）。
FAKE_TOKVAL = hashlib.sha256(b"drama-test-6").hexdigest()[:20]
PEM_HEAD = "-----BEGIN " + "RSA PRIVATE " + "KEY-----"

results = []


def chk(name, ok, detail=""):
    results.append((name, bool(ok), detail))


def scan_text_samples(samples):
    d = Path(tempfile.mkdtemp(prefix="sec_test_"))
    try:
        for fn, body in samples.items():
            p = d / fn
            p.parent.mkdir(parents=True, exist_ok=True)
            p.write_text(body, encoding="utf-8")
        return ss.scan_tree(d)
    finally:
        shutil.rmtree(d, ignore_errors=True)


def n_of(res):
    return len(res["hits"])


# ================================================================ 1. 该报的必须报
# 先把「本机凭证指纹」猴补成运行时假值：测这条规则又不用真密钥。
_orig_real_secrets = ss.real_secrets
ss.real_secrets = lambda: {"FAKE_RH_KEY": FAKE_HEX}

res = scan_text_samples({"leak.py": "KEY = '" + FAKE_HEX + "'\n"})
chk("本机真实凭证值出现在代码里 → 命中", n_of(res) == 1, f"实得 {n_of(res)}")

# ★ 回归（防削弱）：纯 32 位 hex 的凭证**必须**被比对到。
#   （踩过一次：用「纯 hex = 哈希」把凭证指纹过滤掉了 → 只比对到 1 个值，防护形同虚设）
chk("纯 32 位 hex 的本机凭证不会被当成「哈希」放行",
    not ss._obvious_placeholder(FAKE_HEX), "被 _obvious_placeholder 误挡")

ss.real_secrets = lambda: {}                               # 后面几条测通用规则，先清空指纹

res = scan_text_samples({"leak2.py": "k = '" + FAKE_SK + "'"})
chk("sk- 密钥（别的平台）→ 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

res = scan_text_samples({"leak3.py": "k = '" + FAKE_RH + "'"})
chk("rh- 密钥 → 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

res = scan_text_samples({"leak4.py": "headers = {'Authorization': '" + FAKE_BEARER + "'}"})
chk("HTTP Bearer 令牌 → 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

res = scan_text_samples({"leak5.json": '{"project_provider_setting": "' + FAKE_JSON + '"}'})
chk("JSON 里的疑似密钥字段 → 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

res = scan_text_samples({"leak6.py": 'TOKEN = "' + FAKE_TOKVAL + '"'})
chk("无引号/带引号的长 token 字段 → 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

res = scan_text_samples({"leak7.md": PEM_HEAD + "\nMIIE..."})
chk("私钥块 → 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

# 凭证文件本体：不管内容是什么，出现在包里就是问题
res = scan_text_samples({"sub/keys.env": "A=1\n"})
chk("凭证文件本体（keys.env）→ 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

res = scan_text_samples({"certs/server.pem": "not really a cert\n"})
chk("凭证文件本体（*.pem）→ 命中", n_of(res) >= 1, f"实得 {n_of(res)}")

# ================================================================ 2. 不该报的绝不能报
# 这四条里每一条都是**真实踩过的误报**，留着当回归。
res = scan_text_samples({"url.py": 'BASE = "[项目内资源]"'})
chk("端点 URL（http/https）不报", n_of(res) == 0, f"实得 {n_of(res)}")

res = scan_text_samples({"ph.py": 'key = "sk-xxxxxxxxxxxxxxxx"\n'
                                 'key2 = "<你的 key>"\n'
                                 'key3 = os.environ["项目模型_project_provider_setting"]\n'})
chk("占位符 / 环境变量名 / sk-xxxx 不报", n_of(res) == 0, f"实得 {n_of(res)}")

res = scan_text_samples({"code.py": "project_provider_setting = get_project_provider_setting(args.project_provider_setting)\n"
                                   "token = self.token\n"
                                   "password = os.environ.get('X')\n"})
chk("等号右边是**代码**（属性路径/函数调用/变量名）不报",
    n_of(res) == 0, f"实得 {n_of(res)}")

res = scan_text_samples({"hex.json": '{"sha256": "' + FAKE_HEX + '",\n'
                                     ' "audio": "' + FAKE_HEX + '.wav"}'})
chk("纯 sha256 校验值 / 内容哈希文件名不报", n_of(res) == 0, f"实得 {n_of(res)}")

res = scan_text_samples({"rel.md": "python tools/scan_secrets.py --all-zips\n"})
chk("普通命令与相对路径不报", n_of(res) == 0, f"实得 {n_of(res)}")

# ================================================================ 3. 自指陷阱
ss.real_secrets = _orig_real_secrets                        # 恢复真实指纹再扫自己
res = ss.scan_tree(SKILL)
chk("扫描器不报自己（整个 skill 目录命中必须为 0）", n_of(res) == 0,
    "、".join(f"{h['rel']}:{h['line']}" for h in res["hits"][:5]) or "干净")
chk("运行时确实比对了本机凭证（>0，否则等于没比）", res.get("n_secrets", 0) > 0,
    f"n_secrets={res.get('n_secrets')}（此机没配凭证时为 0，属正常）")

# ================================================================ 4. zip 模式
d = Path(tempfile.mkdtemp(prefix="sec_zip_"))
try:
    leakf = d / "leak.py"
    leakf.write_text("k = '" + FAKE_SK + "'\n", encoding="utf-8")
    cleanf = d / "clean.py"
    cleanf.write_text("print('hello')\n", encoding="utf-8")
    enver = d / "keys.env"
    enver.write_text("X=1\n", encoding="utf-8")
    z_dirty = d / "dirty.zip"
    with zipfile.ZipFile(z_dirty, "w") as z:
        z.write(leakf, "references/leak.py")
    z_env = d / "env.zip"
    with zipfile.ZipFile(z_env, "w") as z:
        z.write(enver, "keys.env")
    z_clean = d / "clean.zip"
    with zipfile.ZipFile(z_clean, "w") as z:
        z.write(cleanf, "references/clean.py")
    r1, r2, r3 = ss.scan_zip(z_dirty), ss.scan_zip(z_env), ss.scan_zip(z_clean)
    chk("zip 模式：含密钥的包被抓到", n_of(r1) >= 1, f"实得 {n_of(r1)}")
    chk("zip 模式：包里的 keys.env 被抓到", n_of(r2) >= 1, f"实得 {n_of(r2)}")
    chk("zip 模式：干净的包不误报", n_of(r3) == 0, f"实得 {n_of(r3)}")
finally:
    shutil.rmtree(d, ignore_errors=True)

# ================================================================ 5. 不回显本体
res = scan_text_samples({"leak8.py": "k = '" + FAKE_SK + "'"})
blob = str(res)
chk("报告里不回显密钥本体（只给掩码）", FAKE_SK not in blob or n_of(res) == 0,
    "报告里出现了完整密钥！")

# ================================================================ 输出
print("=" * 78)
print("tools/scan_secrets.py 对抗测试")
print("=" * 78)
for name, ok, detail in results:
    print(f"  [{'PASS' if ok else 'FAIL'}] {name}" + (f"  · {detail}" if detail else ""))
bad = [r for r in results if not r[1]]
print("-" * 78)
print(f"结果：{len(results) - len(bad)} 通过 / {len(bad)} 失败")
for name, _, detail in bad:
    print(f"  ✗ {name}  {detail}")
sys.exit(1 if bad else 0)
